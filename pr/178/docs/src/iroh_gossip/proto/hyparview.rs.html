22/// Input event for HyParView
23#[derive(Debug)]
24pub enum InEvent<PI> {
25    /// A [`Message`] was received from a peer.
26    RecvMessage(PI, Message<PI>),
27    /// A timer has expired.
28    TimerExpired(Timer<PI>),
29    /// A peer was disconnected on the IO layer.
30    PeerDisconnected(PI),
31    /// Send a join request to a peer.
32    RequestJoin(PI),
33    /// Update the peer data that is transmitted on join requests.
34    UpdatePeerData(PeerData),
35    /// Quit the swarm, informing peers about us leaving.
36    Quit,
37}

39/// Output event for HyParView

41pub enum OutEvent<PI> {

75/// Messages that we can send and receive from peers within the topic.
76#[derive(Debug, Serialize, Deserialize, Clone, PartialEq, Eq)]
77pub enum Message<PI> {
78    /// Sent to a peer if you want to join the swarm
79    Join(Option<PeerData>),
80    /// When receiving Join, ForwardJoin is forwarded to the peer's ActiveView to introduce the
81    /// new member.
82    ForwardJoin(ForwardJoin<PI>),
83    /// A shuffle request is sent occasionally to re-shuffle the PassiveView with contacts from
84    /// other peers.
85    Shuffle(Shuffle<PI>),
86    /// Peers reply to [`Message::Shuffle`] requests with a random peers from their active and
87    /// passive views.
88    ShuffleReply(ShuffleReply<PI>),
89    /// Request to add sender to an active view of recipient. If [`Neighbor::priority`] is
90    /// [`Priority::High`], the request cannot be denied.
91    Neighbor(Neighbor),
92    /// Request to disconnect from a peer.
93    /// If [`Disconnect::alive`] is true, the other peer is not shutting down, so it should be
94    /// added to the passive set.
95    Disconnect(Disconnect),
96}

98/// The time-to-live for this message.
99///
100/// Each time a message is forwarded, the `Ttl` is decreased by 1. If the `Ttl` reaches 0, it
101/// should not be forwarded further.
102#[derive(From, Sub, Eq, PartialEq, Clone, Debug, Copy, Serialize, Deserialize)]
103pub struct Ttl(pub u16);
104impl Ttl {
105    pub fn expired(&self) -> bool {
106        *self == Ttl(0)
107    }
108    pub fn next(&self) -> Ttl {
109        Ttl(self.0.saturating_sub(1))
110    }
111}

326    pub fn handle(&mut self, event: InEvent<PI>, io: &mut impl IO<PI>) {
327        match event {
328            InEvent::RecvMessage(from, message) => self.handle_message(from, message, io),
329            InEvent::TimerExpired(timer) => match timer {
330                Timer::DoShuffle => self.handle_shuffle_timer(io),
331                Timer::PendingNeighborRequest(peer, id) => {
332                    self.handle_pending_neighbor_timer(peer, id, io)
333                }
334                Timer::PendingJoin(peer, id) => {
335                    // The peer was not in the topic yet, or the join's connection died.
336                    if self
337                        .pending_joins
338                        .get(&peer)
339                        .is_some_and(|(_, last)| *last == id)
340                    {
341                        self.retry_join(peer, io);
342                    }
343                }
344            },
345            InEvent::PeerDisconnected(peer) => self.handle_connection_closed(peer, io),
346            InEvent::RequestJoin(peer) => self.handle_join(peer, io),
347            InEvent::UpdatePeerData(data) => {
348                self.me_data = Some(data);
349            }
350            InEvent::Quit => self.handle_quit(io),
351        }
352
353        // this will only happen on the first call
354        if !self.shuffle_scheduled {
355            io.push(OutEvent::ScheduleTimer(
356                self.config.shuffle_interval,
357                Timer::DoShuffle,
358            ));
359            self.shuffle_scheduled = true;
360        }
361
362        #[cfg(test)]
363        self.check_invariants();
364    }

366    /// Panics if the per-peer bookkeeping disagrees with the views.
367    ///
368    /// Runs after every event in the unit tests. Every leak fixed so far
369    /// was an entry that outlived its peer's place in a view.
370    #[cfg(test)]
371    fn check_invariants(&self) {
372        let in_a_view =
373            |peer: &PI| self.active_view.contains(peer) || self.passive_view.contains(peer);
374        assert!(
375            self.active_view.len() <= self.config.active_view_capacity,
376            "active view over capacity: {:?}",
377            self.active_view
378        );
379        assert!(
380            self.passive_view.len() <= self.config.passive_view_capacity,
381            "passive view over capacity: {:?}",
382            self.passive_view
383        );
384        assert!(!in_a_view(&self.me), "we are in our own view");
385        for peer in self.active_view.iter() {
386            assert!(
387                !self.passive_view.contains(peer),
388                "{peer:?} is in both views"
389            );
390        }
391        for peer in self.peer_data.keys() {
392            // A forwarded join stores the peer's data while our request to it is out.
393            assert!(
394                in_a_view(peer) || self.pending_neighbor_requests.contains_key(peer),
395                "data kept for {peer:?}, which is in no view and not asked"
396            );
397        }
398        for peer in self.alive_disconnect_peers.iter() {
399            assert!(
400                self.passive_view.contains(peer),
401                "{peer:?} marked alive but not passive"
402            );
403        }
404    }

406    fn handle_message(&mut self, from: PI, message: Message<PI>, io: &mut impl IO<PI>) {
407        let is_disconnect = matches!(message, Message::Disconnect(Disconnect { .. }));
408        if !is_disconnect && !self.active_view.contains(&from) {
409            self.stats.total_connections += 1;
410        }
411        match message {
412            Message::Join(data) => self.on_join(from, data, io),
413            Message::ForwardJoin(details) => self.on_forward_join(from, details, io),
414            Message::Shuffle(details) => self.on_shuffle(from, details, io),
415            Message::ShuffleReply(details) => self.on_shuffle_reply(details, io),
416            Message::Neighbor(details) => self.on_neighbor(from, details, io),
417            Message::Disconnect(details) => self.on_disconnect(from, details, io),
418        }
419
420        // Disconnect from passive nodes right after receiving a message, a
421        // `Disconnect` included.
422        self.release_if_unused(from, io);
423    }
424
425    /// Disconnects `peer` unless it is a neighbor or we wait for its reply.
426    ///
427    /// The reply to our join or `Neighbor` can be on its way, and a closed
428    /// connection would lose it.
429    fn release_if_unused(&self, peer: PI, io: &mut impl IO<PI>) {
430        if !self.active_view.contains(&peer)
431            && !self.pending_neighbor_requests.contains_key(&peer)
432            && !self.pending_joins.contains_key(&peer)
433        {
434            io.push(OutEvent::DisconnectPeer(peer));
435        }
436    }

438    fn handle_join(&mut self, peer: PI, io: &mut impl IO<PI>) {
439        // A neighbor's reply cannot get lost. Tracking it would rejoin the peer when it leaves.
440        if self.active_view.contains(&peer) {
441            io.push(OutEvent::SendMessage(
442                peer,
443                Message::Join(self.me_data.clone()),
444            ));
445        } else {
446            self.pending_joins.entry(peer).or_insert((0, 0));
447            self.send_join(peer, io);
448        }
449    }
450
451    /// Sends a pending join to `peer`, and arms its timeout.
452    fn send_join(&mut self, peer: PI, io: &mut impl IO<PI>) {
453        let id = self.next_join;
454        self.next_join += 1;
455        if let Some((_, last)) = self.pending_joins.get_mut(&peer) {
456            *last = id;
457        }
458        io.push(OutEvent::SendMessage(
459            peer,
460            Message::Join(self.me_data.clone()),
461        ));
462        io.push(OutEvent::ScheduleTimer(
463            self.config.neighbor_request_timeout,
464            Timer::PendingJoin(peer, id),
465        ));
466    }

468    /// Sends a pending join again, or gives up on it after [`JOIN_RETRIES`].
469    fn retry_join(&mut self, peer: PI, io: &mut impl IO<PI>) {
470        let Some((retries, _)) = self.pending_joins.get_mut(&peer) else {
471            return;
472        };
473        if *retries < JOIN_RETRIES {
474            *retries += 1;
475            self.send_join(peer, io);
476        } else {
477            self.pending_joins.remove(&peer);
478            self.release_if_unused(peer, io);
479        }
480    }

482    /// We received a disconnect message.
483    fn on_disconnect(&mut self, peer: PI, details: Disconnect, io: &mut impl IO<PI>) {
484        self.pending_neighbor_requests.remove(&peer);
485        if self.active_view.contains(&peer) {
486            self.remove_active(
487                &peer,
488                RemovalReason::DisconnectReceived {
489                    is_alive: details.alive,
490                },
491                io,
492            );
493        } else if details.alive && self.passive_view.contains(&peer) {
494            self.alive_disconnect_peers.insert(peer);
495        }
496    }
497
498    /// A connection was closed by the peer.
499    fn handle_connection_closed(&mut self, peer: PI, io: &mut impl IO<PI>) {
500        let requested = self.pending_neighbor_requests.remove(&peer).is_some();
501        if self.active_view.contains(&peer) {
502            self.remove_active(&peer, RemovalReason::ConnectionClosed, io);
503        } else {
504            if !self.alive_disconnect_peers.remove(&peer) {
505                self.passive_view.remove(&peer);
506                self.peer_data.remove(&peer);
507            }
508            // The request failed with its dial, long before its timer fires.
509            if requested {
510                self.refill_active_from_passive(&[&peer], io);
511            }
512        }
513        // The join may have gone out on the closed connection.
514        self.retry_join(peer, io);
515    }

517    fn handle_quit(&mut self, io: &mut impl IO<PI>) {
518        for peer in self.active_view.clone().into_iter() {
519            self.active_view.remove(&peer);
520            self.send_disconnect(peer, false, io);
521        }
522        self.pending_joins.clear();
523        self.pending_neighbor_requests.clear();
524        self.passive_view = IndexSet::new();
525        self.peer_data.clear();
526        self.alive_disconnect_peers.clear();
527    }

529    fn send_disconnect(&mut self, peer: PI, alive: bool, io: &mut impl IO<PI>) {
530        // Before disconnecting, send a `ShuffleReply` with some of our nodes to
531        // prevent the other node from running out of connections. This is especially
532        // relevant if the other node just joined the swarm.
533        self.send_shuffle_reply(
534            peer,
535            self.config.shuffle_active_view_count + self.config.shuffle_passive_view_count,
536            io,
537        );
538        let message = Message::Disconnect(Disconnect {
539            alive,
540            _respond: false,
541        });
542        io.push(OutEvent::SendMessage(peer, message));
543        io.push(OutEvent::DisconnectPeer(peer));
544    }

546    fn on_join(&mut self, peer: PI, data: Option<PeerData>, io: &mut impl IO<PI>) {
547        // Drop a neighbor request we may still have pending to this peer. The peer
548        // is joining, usually after a restart, so it has no memory of a request we
549        // sent it before and will not answer it. While that request is pending,
550        // `send_neighbor` would send nothing, and the Join would go unanswered.
551        // The request and the Join can also cross in flight: another of the
552        // peer's bootstrap peers forwarded its join to us, and we sent the peer a
553        // request before its own Join reached us. Then the peer gets one Neighbor
554        // more than needed and takes it as the reply to its own.
555        self.pending_neighbor_requests.remove(&peer);
556        // "A node that receives a join request will start by adding the new
557        // node to its active view, even if it has to drop a random node from it. (6)"
558        self.add_active(peer, data.clone(), Priority::High, true, io);
559
560        // "The contact node c will then send to all other nodes in its active view a ForwardJoin
561        // request containing the new node identifier. Associated to the join procedure,
562        // there are two configuration parameters, named Active Random Walk Length (ARWL),
563        // that specifies the maximum number of hops a ForwardJoin request is propagated,
564        // and Passive Random Walk Length (PRWL), that specifies at which point in the walk the node
565        // is inserted in a passive view. To use these parameters, the ForwardJoin request carries
566        // a “time to live” field that is initially set to ARWL and decreased at every hop. (7)"
567        let ttl = self.config.active_random_walk_length;
568        let peer_info = PeerInfo { id: peer, data };
569        for node in self.active_view.iter_without(&peer) {
570            let message = Message::ForwardJoin(ForwardJoin {
571                peer: peer_info.clone(),
572                ttl,
573            });
574            io.push(OutEvent::SendMessage(*node, message));
575        }
576    }

578    fn on_forward_join(&mut self, sender: PI, message: ForwardJoin<PI>, io: &mut impl IO<PI>) {
579        let peer_id = message.peer.id;
580        // If the peer is already in our active view, we renew our neighbor relationship.
581        if self.active_view.contains(&peer_id) {
582            self.insert_peer_info(message.peer, io);
583            self.send_neighbor(peer_id, Priority::High, io);
584        }
585        // "i) If the time to live is equal to zero or if the number of nodes in p’s active view is equal to one,
586        // it will add the new node to its active view (7)"
587        else if message.ttl.expired() || self.active_view.len() <= 1 {
588            self.insert_peer_info(message.peer, io);
589            // Modification from paper: Instead of adding the peer directly to our active view,
590            // we only send the Neighbor message. We will add the peer to our active view once we receive a
591            // reply from our neighbor.
592            // This prevents us adding unreachable peers to our active view.
593            self.send_neighbor(peer_id, Priority::High, io);
594        } else {
595            // "ii) If the time to live is equal to PRWL, p will insert the new node into its passive view"
596            if message.ttl == self.config.passive_random_walk_length {
597                self.add_passive(peer_id, message.peer.data.clone(), io);
598            }
599            // "iii) The time to live field is decremented."
600            // "iv) If, at this point, n has not been inserted
601            // in p’s active view, p will forward the request to a random node in its active view
602            // (different from the one from which the request was received)."
603            if !self.active_view.contains(&peer_id)
604                && !self.pending_neighbor_requests.contains_key(&peer_id)
605            {
606                match self
607                    .active_view
608                    .pick_random_without(&[&sender], &mut self.rng)
609                {
610                    None => {
611                        unreachable!("if the peer was not added, there are at least two peers in our active view.");
612                    }
613                    Some(next) => {
614                        let message = Message::ForwardJoin(ForwardJoin {
615                            peer: message.peer,
616                            ttl: message.ttl.next(),
617                        });
618                        io.push(OutEvent::SendMessage(*next, message));
619                    }
620                }
621            }
622        }
623    }

625    fn on_neighbor(&mut self, from: PI, details: Neighbor, io: &mut impl IO<PI>) {
626        // A `Neighbor` is a reply if we have a request out to the peer, and a
627        // request otherwise. See `send_neighbor` for the full picture.
628        let is_reply = self.pending_neighbor_requests.remove(&from).is_some();
629        let do_reply = !is_reply;
630        // "A node q that receives a high priority neighbor request will always accept the request, even
631        // if it has to drop a random member from its active view (again, the member that is dropped will
632        // receive a Disconnect notification). If a node q receives a low priority Neighbor request, it will
633        // only accept the request if it has a free slot in its active view, otherwise it will refuse the request."
634        if !self.add_active(from, details.data, details.priority, do_reply, io) {
635            self.send_disconnect(from, true, io);
636            // `add_active` stored the peer's data before it refused the peer.
637            self.forget_peer(&from);
638        }
639    }

661    /// Handle a [`Message::Shuffle`]
662    ///
663    /// > A node q that receives a Shuffle request will first decrease its time to live. If the time
664    /// > to live of the message is greater than zero and the number of nodes in q’s active view is
665    /// > greater than 1, the node will select a random node from its active view, different from the
666    /// > one he received this shuffle message from, and simply forwards the Shuffle request.
667    /// > Otherwise, node q accepts the Shuffle request and send back (p.8)
668    fn on_shuffle(&mut self, from: PI, shuffle: Shuffle<PI>, io: &mut impl IO<PI>) {
669        if shuffle.ttl.expired() || self.active_view.len() <= 1 {
670            let len = shuffle.nodes.len();
671            for node in shuffle.nodes {
672                self.add_passive(node.id, node.data, io);
673            }
674            self.send_shuffle_reply(shuffle.origin, len, io);
675            // The origin is usually no neighbor, and the reply its only use.
676            self.release_if_unused(shuffle.origin, io);
677        } else if let Some(node) = self
678            .active_view
679            .pick_random_without(&[&shuffle.origin, &from], &mut self.rng)
680        {
681            let message = Message::Shuffle(Shuffle {
682                origin: shuffle.origin,
683                nodes: shuffle.nodes,
684                ttl: shuffle.ttl.next(),
685            });
686            io.push(OutEvent::SendMessage(*node, message));
687        }
688    }

690    fn send_shuffle_reply(&mut self, to: PI, len: usize, io: &mut impl IO<PI>) {
691        let mut nodes = self.passive_view.shuffled_and_capped(len, &mut self.rng);
692        // If we don't have enough passive nodes for the expected length, we fill with
693        // active nodes.
694        if nodes.len() < len {
695            nodes.extend(
696                self.active_view
697                    .shuffled_and_capped(len - nodes.len(), &mut self.rng),
698            );
699        }
700        let nodes = nodes.into_iter().map(|id| self.peer_info(&id));
701        let message = Message::ShuffleReply(ShuffleReply {
702            nodes: nodes.collect(),
703        });
704        io.push(OutEvent::SendMessage(to, message));
705    }

708        for node in message.nodes {
709            self.add_passive(node.id, node.data, io);
710        }
711        self.refill_active_from_passive(&[], io);

714    fn handle_shuffle_timer(&mut self, io: &mut impl IO<PI>) {
715        if let Some(node) = self.active_view.pick_random(&mut self.rng) {
716            let active = self.active_view.shuffled_without_and_capped(
717                &[node],
718                self.config.shuffle_active_view_count,
719                &mut self.rng,
720            );
721            let passive = self.passive_view.shuffled_without_and_capped(
722                &[node],
723                self.config.shuffle_passive_view_count,
724                &mut self.rng,
725            );
726            let nodes = active
727                .iter()
728                .chain(passive.iter())
729                .map(|id| self.peer_info(id));
730            let me = PeerInfo {
731                id: self.me,
732                data: self.me_data.clone(),
733            };
734            let nodes = nodes.chain([me]);
735            let message = Shuffle {
736                origin: self.me,
737                nodes: nodes.collect(),
738                ttl: self.config.shuffle_random_walk_length,
739            };
740            io.push(OutEvent::SendMessage(*node, Message::Shuffle(message)));
741        }
742        io.push(OutEvent::ScheduleTimer(
743            self.config.shuffle_interval,
744            Timer::DoShuffle,
745        ));
746    }

756    /// Add a peer to the passive view.
757    ///
758    /// If the passive view is full, it will first remove a random peer and then insert the new peer.
759    /// If a peer is currently in the active view it will not be added.
760    fn add_passive(&mut self, peer: PI, data: Option<PeerData>, io: &mut impl IO<PI>) {
761        self.insert_peer_info((peer, data).into(), io);
762        if self.active_view.contains(&peer) || self.passive_view.contains(&peer) || peer == self.me
763        {
764            return;
765        }
766        if self.passive_is_full() {
767            if let Some(evicted) = self.passive_view.remove_random(&mut self.rng) {
768                self.forget_peer(&evicted);
769            }
770        }
771        self.passive_view.insert(peer);
772    }

774    /// Drops the metadata we hold for a peer, unless it is still in a view.
775    ///
776    /// `peer_data` and `alive_disconnect_peers` describe peers in a view, so
777    /// every path out of a view has to clear them. The membership check is for
778    /// callers that cannot tell: a neighbor request can time out after the peer
779    /// joined the active view through a forwarded join.
780    fn forget_peer(&mut self, peer: &PI) {
781        if self.active_view.contains(peer) || self.passive_view.contains(peer) {
782            return;
783        }
784        self.peer_data.remove(peer);
785        self.alive_disconnect_peers.remove(peer);
786    }

788    /// Remove a peer from the active view.
789    ///
790    /// If `reason` is [`RemovalReason::Random`], a [`Disconnect`] message will be sent to the peer.
791    fn remove_active(&mut self, peer: &PI, reason: RemovalReason, io: &mut impl IO<PI>) {
792        if let Some(idx) = self.active_view.get_index_of(peer) {
793            let removed_peer = self.remove_active_by_index(idx, reason, io).unwrap();
794            self.refill_active_from_passive(&[&removed_peer], io);
795        }
796    }

798    fn refill_active_from_passive(&mut self, skip_peers: &[&PI], io: &mut impl IO<PI>) {
799        // A pending reply is to a peer already in the active view; count it once.
800        let pending_outside = self
801            .pending_neighbor_requests
802            .keys()
803            .filter(|peer| !self.active_view.contains(*peer))
804            .count();
805        if self.active_view.len() + pending_outside >= self.config.active_view_capacity {
806            return;
807        }
808        // "When a node p suspects that one of the nodes present in its active view has failed
809        // (by either disconnecting or blocking), it selects a random node q from its passive view and
810        // attempts to establish a TCP connection with q. If the connection fails to establish,
811        // node q is considered failed and removed from p’s passive view; another node q′ is selected
812        // at random and a new attempt is made. The procedure is repeated until a connection is established
813        // with success." (p7)
814        let mut skip_peers = skip_peers.to_vec();
815        skip_peers.extend(self.pending_neighbor_requests.keys());
816
817        if let Some(node) = self
818            .passive_view
819            .pick_random_without(&skip_peers, &mut self.rng)
820            .copied()
821        {
822            let priority = match self.active_view.is_empty() {
823                true => Priority::High,
824                false => Priority::Low,
825            };
826            self.send_neighbor(node, priority, io);
827        };
828    }

830    /// Handles a `Neighbor` to `peer` that was not answered in time.
831    ///
832    /// For a request, the peer is taken as failed and dropped from the passive
833    /// view. For a reply, which is never answered, the peer is in the active
834    /// view and keeps its data; the entry is cleared, and the refill below
835    /// only acts if the active view has room. A timer whose `id` is not the
836    /// pending one belongs to an earlier `Neighbor`, and does nothing.
837    fn handle_pending_neighbor_timer(&mut self, peer: PI, id: u64, io: &mut impl IO<PI>) {
838        if self.pending_neighbor_requests.get(&peer) == Some(&id) {
839            self.pending_neighbor_requests.remove(&peer);
840            self.passive_view.remove(&peer);
841            self.forget_peer(&peer);
842            self.release_if_unused(peer, io);
843            self.refill_active_from_passive(&[], io);
844        }
845    }

847    fn remove_active_by_index(
848        &mut self,
849        peer_index: usize,
850        reason: RemovalReason,
851        io: &mut impl IO<PI>,
852    ) -> Option<PI> {
853        if let Some(peer) = self.active_view.remove_index(peer_index) {
854            io.push(OutEvent::EmitEvent(Event::NeighborDown(peer)));
855
856            match reason {
857                // send a disconnect message, then close connection.
858                RemovalReason::Random => self.send_disconnect(peer, true, io),
859                // close connection without sending anything further.
860                RemovalReason::DisconnectReceived { is_alive: _ } => {
861                    io.push(OutEvent::DisconnectPeer(peer))
862                }
863                RemovalReason::ConnectionClosed => io.push(OutEvent::DisconnectPeer(peer)),
864            }
865
866            let keep_as_passive = match reason {
867                // keep alive if previously marked as alive.
868                RemovalReason::ConnectionClosed => self.alive_disconnect_peers.remove(&peer),
869                // keep alive if other peer said to be still alive.
870                RemovalReason::DisconnectReceived { is_alive } => is_alive,
871                // keep alive (only we are removing for now)
872                RemovalReason::Random => true,
873            };
874
875            if keep_as_passive {
876                let data = self.peer_data.remove(&peer);
877                self.add_passive(peer, data, io);
878                // mark peer as alive, so it doesn't get removed from the passive view if the conn closes.
879                if !matches!(reason, RemovalReason::ConnectionClosed) {
880                    self.alive_disconnect_peers.insert(peer);
881                }
882            } else {
883                self.forget_peer(&peer);
884            }
885            debug!(other = ?peer, "removed from active view, reason: {reason:?}");
886            Some(peer)
887        } else {
888            None
889        }
890    }

899    /// Add a peer to the active view.
900    ///
901    /// If the active view is currently full, a random peer will be removed first.
902    /// Sends a Neighbor message to the peer. If high_priority is true, the peer
903    /// may not deny the Neighbor request.
904    fn add_active(
905        &mut self,
906        peer: PI,
907        data: Option<PeerData>,
908        priority: Priority,
909        reply: bool,
910        io: &mut impl IO<PI>,
911    ) -> bool {
912        if peer == self.me {
913            return false;
914        }
915        self.insert_peer_info((peer, data).into(), io);
916        if self.active_view.contains(&peer) {
917            if reply {
918                self.send_neighbor(peer, priority, io);
919            }
920            return true;
921        }
922        match (priority, self.active_is_full()) {
923            (Priority::High, is_full) => {
924                if is_full {
925                    self.free_random_slot_in_active_view(io);
926                }
927                self.add_active_unchecked(peer, Priority::High, reply, io);
928                true
929            }
930            (Priority::Low, false) => {
931                self.add_active_unchecked(peer, Priority::Low, reply, io);
932                true
933            }
934            (Priority::Low, true) => false,
935        }
936    }

938    fn add_active_unchecked(
939        &mut self,
940        peer: PI,
941        priority: Priority,
942        reply: bool,
943        io: &mut impl IO<PI>,
944    ) {
945        self.passive_view.remove(&peer);
946        self.pending_joins.remove(&peer);
947        self.alive_disconnect_peers.remove(&peer);
948        if self.active_view.insert(peer) {
949            debug!(other = ?peer, "add to active view");
950            io.push(OutEvent::EmitEvent(Event::NeighborUp(peer)));
951            if reply {
952                self.send_neighbor(peer, priority, io);
953            }
954        }
955    }

957    /// Sends a `Neighbor` unless one is pending for the peer, and arms its timer.
958    ///
959    /// `Neighbor` is both the request to become neighbors and the reply that
960    /// accepts it. `on_neighbor` tells them apart by `pending_neighbor_requests`:
961    /// a `Neighbor` from a peer we have one pending for is the reply, any other
962    /// is a request. Replies are recorded as pending too. That is what stops two
963    /// peers from answering each other forever when a request is duplicated: the
964    /// extra `Neighbor` is taken as the reply and not answered. The entry for a
965    /// reply is cleared by the timer alone.
966    fn send_neighbor(&mut self, peer: PI, priority: Priority, io: &mut impl IO<PI>) {
967        if let hash_map::Entry::Vacant(entry) = self.pending_neighbor_requests.entry(peer) {
968            let id = self.next_neighbor_request;
969            self.next_neighbor_request += 1;
970            entry.insert(id);
971            let message = Message::Neighbor(Neighbor {
972                priority,
973                data: self.me_data.clone(),
974            });
975            io.push(OutEvent::SendMessage(peer, message));
976            io.push(OutEvent::ScheduleTimer(
977                self.config.neighbor_request_timeout,
978                Timer::PendingNeighborRequest(peer, id),
979            ));
980        }
981    }
