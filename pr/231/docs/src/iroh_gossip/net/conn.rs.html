26/// How long a connection stays open once neither side has a stream on it.
27///
28/// A topic that uses the peer keeps a stream open. Each side times the grace on
29/// its own, so a stream that we open as the peer closes loses its message.
30#[cfg(not(test))]
31const RETIRE_GRACE: Duration = Duration::from_secs(10);
32/// The e2e tests build without `cfg(test)`, so they see 10 s.
33#[cfg(test)]
34pub(super) const RETIRE_GRACE: Duration = Duration::from_secs(5);
35
36/// How long a replaced connection may write its current message before it hands it over.
37///
38/// The new connection of the peer waits for it, so that we write the messages
39/// in order. A stream header that started is written in full first. Only
40/// the write timeout bounds it, which counts the time without progress.
41#[cfg(not(test))]
42const HANDOVER_LIMIT: Duration = Duration::from_secs(5);
43#[cfg(test)]
44pub(super) const HANDOVER_LIMIT: Duration = Duration::from_secs(1);
45
46/// The code of a close for a peer whose send queue had no room for a control message.
47pub(super) const CLOSE_QUEUE_FULL: u32 = 2;
48
49/// The code of a close for a write or a stream open that made no progress in time.
50pub(super) const CLOSE_STALLED: u32 = 3;
51
52/// The code of a close for a connection without streams, see [`RETIRE_GRACE`](super::RETIRE_GRACE).
53pub(super) const CLOSE_RETIRED: u32 = 4;

55/// The maximum number of stream headers that a [`RecvLoop`] reads at the same time.
56///
57/// At this limit, the loop accepts no new stream until a header is complete.
58/// A peer that keeps many partial headers thus delays only its own streams.
59pub(super) const MAX_PENDING_HEADERS: usize = 16;

83/// Returns a finished task's output, or `None` if the runtime cancelled the task.
84///
85/// A task cancelled at runtime shutdown is not a bug, so it must not panic the actor (#140).
86/// A panic in the task panics the caller with the same payload.
87pub(super) fn task_output<T>(res: Result<T, task::JoinError>) -> Option<T> {
88    match res {
89        Ok(output) => Some(output),
90        Err(err) if err.is_cancelled() => None,
91        #[cfg(not(target_family = "wasm"))]
92        Err(err) => std::panic::resume_unwind(err.into_panic()),
93        #[cfg(target_family = "wasm")]
94        Err(err) => panic!("task failed: {err}"),
95    }
96}

98/// Returns once the connection had no stream in either direction for [`RETIRE_GRACE`].
99pub(super) async fn retire(mut ours: watch::Receiver<bool>, mut theirs: watch::Receiver<bool>) {
100    loop {
101        let ours_open = *ours.borrow_and_update();
102        let theirs_open = *theirs.borrow_and_update();
103        tokio::select! {
104            // A stream that opens at the deadline keeps the connection.
105            biased;
106            // A dropped sender would make `changed` return at once.
107            res = ours.changed() => if res.is_err() { std::future::pending().await },
108            res = theirs.changed() => if res.is_err() { std::future::pending().await },
109            () = time::sleep(RETIRE_GRACE), if !ours_open && !theirs_open => return,
110        }
111    }
112}

114/// Runs the I/O of a connection, and closes it on an abort or once it carries no stream.
115///
116/// Returns `true` if the end of the connection does not count as a disconnect.
117/// That is the case when either side retired the connection, or when no topic
118/// used the peer at the moment the connection ended. A connection to a node up
119/// to 0.101 never retires. Without a lease, the connection only receives.
120/// Every connection observes the peer abort signal, independently of whether
121/// it writes.
122// Each argument is state the loops share. A struct would only wrap them.
123#[allow(clippy::too_many_arguments)]
124pub(super) async fn connection_loop(
125    from: PublicKey,
126    conn: Connection,
127    aborted: CancellationToken,
128    queue_lease: Option<QueueLease>,
129    initial_topics: Vec<TopicId>,
130    in_event_tx: mpsc::Sender<InEvent>,
131    max_message_size: usize,
132    handles_close: watch::Sender<bool>,
133) -> Result<bool, ConnectionLoopError> {
134    debug!(side = ?conn.side(), "connection established");
135    let mut handles_close_rx = handles_close.subscribe();
136    // Here, so that the flags outlive the loops. The send flag starts set, as a
137    // new writer can wait for the lock of the queue.
138    let our_streams = watch::Sender::new(queue_lease.is_some());
139    let their_streams = watch::Sender::new(false);
140
141    let queue_closed = queue_lease.as_ref().map(QueueLease::closing);
142    let send_loop = queue_lease.map(|lease| {
143        let flag = our_streams.clone();
144        SendLoop::new(conn.clone(), lease, max_message_size, flag, initial_topics)
145    });
146    let mut recv_loop = RecvLoop::new(
147        from,
148        conn.clone(),
149        in_event_tx,
150        max_message_size,
151        handles_close,
152        their_streams.clone(),
153    );
154
155    let io = async {
156        let recv_fut = recv_loop.run().instrument(error_span!("recv"));
157        tokio::pin!(recv_fut);
158        if let Some(send_loop) = send_loop {
159            tokio::select! {
160                res = send_loop.run().instrument(error_span!("send")) => res?,
161                _ = &mut recv_fut => return Ok(()),
162            }
163        }
164        // We are done sending. The peer may still send on this connection, as two
165        // peers that dialed each other can each use a different one.
166        recv_fut.await;
167        Ok(())
168    };
169    let retirement = async {
170        // An old node would lose its later messages to us after a close, see
171        // `StreamHeader::handles_close`.
172        if handles_close_rx.wait_for(|known| *known).await.is_err() {
173            std::future::pending::<()>().await;
174        }
175        retire(our_streams.subscribe(), their_streams.subscribe()).await;
176    };
177    tokio::select! {
178        biased;
179        // All connections close, including receive-only ones, so the peer
180        // learns it on the connection that it uses. A
181        // node up to 0.101 may then lose its later messages to us (#155). It
182        // missed a control message, so its views differ from ours anyway.
183        _ = aborted.cancelled() => {
184            warn!("the peer is aborted, close the connection");
185            conn.close(CLOSE_QUEUE_FULL.into(), b"send queue full");
186            Ok(false)
187        }
188        res = io => {
189            // A retirement by the peer counts also when it cut one of our writes.
190            let retired_by_peer = matches!(
191                conn.close_reason(),
192                Some(ConnectionError::ApplicationClosed(close))
193                    if close.error_code == CLOSE_RETIRED.into()
194            );
195            // Read now, as a topic may open the queue before the actor sees the end.
196            // A message from then on needs a new dial, not a disconnect.
197            let queue_was_closed = queue_closed.is_some_and(|closed| *closed.borrow());
198            if retired_by_peer || queue_was_closed {
199                Ok(true)
200            } else {
201                res.map(|()| false)
202            }
203        }
204        _ = retirement => {
205            debug!("no stream for a while, close");
206            conn.close(CLOSE_RETIRED.into(), b"no stream");
207            Ok(true)
208        }
209    }
210}

225/// The result of a stream read: the stream, and its next message or its end.
226type ReadResult = (TopicReader, Result<Option<ProtoMessage>, ReadError>);

228impl RecvLoop {
229    pub(super) fn new(
230        remote_endpoint_id: EndpointId,
231        conn: Connection,
232        in_event_tx: mpsc::Sender<InEvent>,
233        max_message_size: usize,
234        handles_close: watch::Sender<bool>,
235        has_streams: watch::Sender<bool>,
236    ) -> Self {
237        Self {
238            remote_endpoint_id,
239            conn,
240            max_message_size,
241            in_event_tx,
242            handles_close,
243            has_streams,
244        }
245    }
246
247    /// Reads the streams of the connection until it closes or the actor stops.
248    ///
249    /// A bad stream loses only itself, so this never fails.
250    pub(super) async fn run(&mut self) {
251        // Each header is read in its own future, so a partial one blocks no other stream.
252        let mut header_futures = FuturesUnordered::new();
253        let mut read_futures = FuturesUnordered::new();
254        // For each topic, the streams in the order the peer opened them. The peer
255        // only opens a new stream for a topic after the previous one ended. So if
256        // a stream waits for the older ones, the order is kept and the wait ends.
257        let mut streams: BTreeMap<(TopicId, StreamId), Order> = BTreeMap::new();
258        let mut conn_is_closed = false;
259        let closed = self.conn.closed();
260        tokio::pin!(closed);
261        while !conn_is_closed || !header_futures.is_empty() || !read_futures.is_empty() {
262            // A partial header counts. Only a broken peer keeps one partial.
263            let open = !header_futures.is_empty() || !read_futures.is_empty();
264            set_if_changed(&self.has_streams, open);
265            tokio::select! {
266                _ = &mut closed, if !conn_is_closed => {
267                    conn_is_closed = true;
268                }
269                stream = self.conn.accept_uni(), if !conn_is_closed && header_futures.len() < MAX_PENDING_HEADERS => {
270                    let stream = match stream {
271                        Ok(stream) => stream,
272                        Err(_) => {
273                            conn_is_closed = true;
274                            continue;
275                        }
276                    };
277                    header_futures.push(TopicReader::accept(stream, self.max_message_size));
278                }
279                Some(state) = header_futures.next(), if !header_futures.is_empty() => {
280                    // A bad header loses only its stream, as a bad frame does.
281                    let state = match state {
282                        Ok(state) => state,
283                        Err(err) => {
284                            debug!("stream header failed: {err:#}");
285                            continue;
286                        }
287                    };
288                    debug!(topic=%state.header.topic.fmt_short(), "stream opened");
289                    if state.header.handles_close {
290                        self.handles_close.send_replace(true);
291                    }
292                    let (topic, id) = (state.header.topic, state.stream.id());
293                    // A stream waits while there is an older stream for the same topic. If
294                    // the header of an older stream arrives late, we read that stream right
295                    // away, and it does not wait for the newer one.
296                    let before = streams.range(..(topic, id)).next_back();
297                    let is_first_for_topic = before.is_none_or(|((t, _), _)| *t != topic);
298                    if is_first_for_topic {
299                        streams.insert((topic, id), Order::Reading);
300                        read_futures.push(state.read_next());
301                    } else {
302                        streams.insert((topic, id), Order::Waiting(state));
303                    }
304                }
305                Some((state, res)) = read_futures.next(), if !read_futures.is_empty() => {
306                    match res {
307                        Ok(Some(msg)) => {
308                            if self.in_event_tx.send(InEvent::RecvMessage(self.remote_endpoint_id, msg)).await.is_err() {
309                                debug!("stop recv loop: actor closed");
310                                break;
311                            }
312                            read_futures.push(state.read_next());
313                            continue;
314                        }
315                        Ok(None) => {
316                            debug!(topic=%state.header.topic.fmt_short(), "stream closed");
317                        }
318                        Err(err) => debug!("recv stream closed with error: {err:#}"),
319                    }
320                    let (topic, id) = (state.header.topic, state.stream.id());
321                    streams.remove(&(topic, id));
322                    // Start reading the next stream for this topic, unless we already do.
323                    let after = streams.range_mut((topic, id)..).next();
324                    let next_for_topic = after.filter(|((t, _), _)| *t == topic);
325                    if let Some((_, order)) = next_for_topic {
326                        if let Order::Waiting(next) = std::mem::replace(order, Order::Reading) {
327                            read_futures.push(next.read_next());
328                        }
329                    }
330                }
331            }
332        }
333        debug!("recv loop closed");
334    }
335}

337/// Whether a stream in a [`RecvLoop`] is read now, or waits for an older stream of the same topic.
338enum Order {
339    Reading,
340    Waiting(TopicReader),
341}

343/// A topic stream that we read: its header, and the buffer for its frames.
344#[derive(Debug)]
345struct TopicReader {
346    stream: RecvStream,
347    header: StreamHeader,
348    buffer: BytesMut,
349    max_message_size: usize,
350}
351
352impl TopicReader {
353    /// Reads the header of a new stream.
354    async fn accept(mut stream: RecvStream, max_message_size: usize) -> Result<Self, ReadError> {
355        let mut buffer = BytesMut::new();
356        let header = StreamHeader::read(&mut stream, &mut buffer, max_message_size).await?;
357        Ok(Self {
358            buffer,
359            max_message_size,
360            stream,
361            header,
362        })
363    }
364
365    /// Reads the next message of the stream, or `None` if it ended gracefully.
366    ///
367    /// Returns `self` together with the result, also when the read failed.
368    async fn read_next(mut self) -> ReadResult {
369        let msg = read_message(&mut self.stream, &mut self.buffer, self.max_message_size).await;
370        let topic = self.header.topic;
371        let msg = msg.map(|msg| msg.map(|message| ProtoMessage { topic, message }));
372        (self, msg)
373    }
374}

376pub(super) struct SendLoop {
377    conn: Connection,
378    streams: HashMap<TopicId, SendStream>,
379    buffer: Vec<u8>,
380    max_message_size: usize,
381    /// A future per stream, which resolves once the peer stopped it or acknowledged all of it.
382    live_streams: FuturesUnordered<BoxFuture<(TopicId, StreamId)>>,
383    queue_lease: QueueLease,
384    /// The topics that used the peer when the writer started, for which it opens streams.
385    initial_topics: Vec<TopicId>,
386    /// Whether we have a stream to the peer.
387    ///
388    /// It is set before an open, and clears once [`Self::live_streams`] is empty.
389    has_streams: watch::Sender<bool>,
390}

392impl SendLoop {
393    pub(super) fn new(
394        conn: Connection,
395        queue_lease: QueueLease,
396        max_message_size: usize,
397        has_streams: watch::Sender<bool>,
398        initial_topics: Vec<TopicId>,
399    ) -> Self {
400        Self {
401            conn,
402            max_message_size,
403            buffer: Default::default(),
404            streams: Default::default(),
405            live_streams: Default::default(),
406            queue_lease,
407            initial_topics,
408            has_streams,
409        }
410    }
411
412    /// Writes the queue until it ends, the lease is revoked, or the connection closes.
413    ///
414    /// Once no queued message waits, it also opens a stream for each initial
415    /// topic. After the queue closed, it waits for the queue to open again.
416    /// Takes `self`, so that the lease drops once it returns.
417    pub(super) async fn run(mut self) -> Result<(), WriteError> {
418        let conn = self.conn.clone();
419        let closed = conn.closed();
420        tokio::pin!(closed);
421        let revoked = self.queue_lease.revoked().clone();
422        // A replaced connection of the peer may still write its current message.
423        let queue = tokio::select! {
424            biased;
425            _ = &mut closed => None,
426            _ = revoked.cancelled() => None,
427            queue = self.queue_lease.lock() => Some(queue),
428        };
429        if let Some(mut queue) = queue {
430            let mut closing = self.queue_lease.closing();
431            loop {
432                // Every pass of the loop starts here, so the flag clears with the last stream.
433                if self.live_streams.is_empty() {
434                    set_if_changed(&self.has_streams, false);
435                }
436                let more = tokio::select! {
437                    biased;
438                    _ = &mut closed => break,
439                    // The next connection takes the rest of the queue.
440                    _ = revoked.cancelled() => break,
441                    // Before the queue: the queue of a busy peer is always ready, and
442                    // would keep ended streams from being collected.
443                    Some((topic, id)) = self.live_streams.next(), if !self.live_streams.is_empty() => {
444                        // The peer stopped the stream, as on a frame over its size limit.
445                        // The next message of the topic opens a new one.
446                        if self.streams.get(&topic).is_some_and(|s| s.id() == id) {
447                            self.streams.remove(&topic);
448                        }
449                        continue;
450                    }
451                    more = queue.wait_next() => more,
452                    // This comes after the queue, so it only runs when the queue is empty. We
453                    // finish our streams and keep waiting on this connection, so the next
454                    // message does not need a dial. Dropping a stream finishes it.
455                    _ = async { closing.wait_for(|closing| *closing).await.ok(); },
456                        if !self.streams.is_empty() || !self.initial_topics.is_empty() =>
457                    {
458                        self.streams.clear();
459                        self.initial_topics.clear();
460                        continue;
461                    }
462                    // After the queue, so that queued messages get the streams of the peer
463                    // first. A quiet topic needs a stream to keep the connection.
464                    Some(topic) = async { self.initial_topics.pop() } => {
465                        match self.open_stream(topic, &revoked).await {
466                            // At the stream limit of the peer, the quiet topics go without.
467                            Err(WriteError::OpenStalled { .. }) => self.initial_topics.clear(),
468                            Err(err) => self.handle_error(topic, err)?,
469                            Ok(_) => {}
470                        }
471                        continue;
472                    }
473                };
474                if !more {
475                    // The actor dropped the queue.
476                    break;
477                }
478                let item = queue.current().expect("next returned a message");
479                match self.open_stream(item.message.topic, &revoked).await {
480                    Ok(true) => {}
481                    // The lease ended before the stream opened. The message stays
482                    // in the queue for the next connection.
483                    Ok(false) => break,
484                    Err(err) => {
485                        self.handle_error(item.message.topic, err)?;
486                        queue.advance();
487                        continue;
488                    }
489                }
490                // A stuck write would keep the queue from the connection that replaced
491                // this one. A message that we stop in the middle stays in the queue,
492                // and the next connection writes it first. Its stream stays, and ends
493                // with the others: the peer reads the messages before it, and drops
494                // the part of this one.
495                let handover = async {
496                    revoked.cancelled().await;
497                    n0_future::time::sleep(HANDOVER_LIMIT).await;
498                };
499                tokio::select! {
500                    biased;
501                    res = self.send(item) => res?,
502                    _ = handover => {
503                        warn!("the replaced connection did not finish its message in time, hand it over");
504                        break;
505                    }
506                }
507                // A topic that we just left needs no stream.
508                if item.message.message.is_disconnect() {
509                    self.initial_topics
510                        .retain(|topic| *topic != item.message.topic);
511                }
512                queue.advance();
513            }
514        }
515        self.streams.clear();
516        while self.live_streams.next().await.is_some() {}
517        set_if_changed(&self.has_streams, false);
518        debug!("send loop closed");
519        Ok(())
520    }
521
522    /// Writes a message on the stream of its topic, and finishes the stream after a `Close`.
523    ///
524    /// If the write fails because the peer stopped the stream, the message is lost.
525    /// Returns the error only if the connection is gone, or if the write made no
526    /// progress in time, which closes the connection.
527    ///
528    /// ## Cancellation safety
529    ///
530    /// This function is not cancellation-safe.
531    async fn send(&mut self, item: &QueuedMessage) -> Result<(), WriteError> {
532        let ProtoMessage { topic, message } = &item.message;
533        let stream = self
534            .streams
535            .get_mut(topic)
536            .expect("the stream was opened before");
537        let len = item.encoded_len();
538        match write_sized_message(
539            stream,
540            message,
541            len,
542            &mut self.buffer,
543            self.max_message_size,
544        )
545        .await
546        {
547            Ok(()) => {
548                if message.is_disconnect() {
549                    // Dropping a stream finishes it.
550                    self.streams.remove(topic);
551                }
552                Ok(())
553            }
554            // Refused before any byte went out, so the stream is intact.
555            Err(WriteError::TooLarge { .. }) => {
556                warn!(max = self.max_message_size, "message too large, dropped");
557                Ok(())
558            }
559            Err(err) => {
560                self.handle_error(*topic, err)?;
561                self.streams.remove(topic);
562                Ok(())
563            }
564        }
565    }
566
567    /// Returns the error if the connection is gone or stalled.
568    ///
569    /// Otherwise only one stream failed, and the loop goes on.
570    fn handle_error(&self, topic: TopicId, err: WriteError) -> Result<(), WriteError> {
571        if matches!(
572            err,
573            WriteError::Stalled { .. } | WriteError::OpenStalled { .. }
574        ) {
575            // A peer that reads nothing blocks every stream, so a new stream would not help.
576            warn!("{err}, close the connection");
577            self.conn.close(CLOSE_STALLED.into(), b"write stalled");
578            Err(err)
579        } else if self.conn.close_reason().is_some() {
580            Err(err)
581        } else {
582            warn!(topic = %topic.fmt_short(), "write failed: {err:#}");
583            Ok(())
584        }
585    }

587    /// Opens the stream of `topic` and writes its header, unless it is open.
588    ///
589    /// Returns `false` if `revoked` ends the lease before the stream opened. A
590    /// header that started is written in full, as a node up to 0.101 drops the
591    /// whole connection on a partial one. An abort of the peer closes the
592    /// connection anyway, and cuts the header.
593    async fn open_stream(
594        &mut self,
595        topic: TopicId,
596        revoked: &CancellationToken,
597    ) -> Result<bool, WriteError> {
598        let hash_map::Entry::Vacant(entry) = self.streams.entry(topic) else {
599            return Ok(true);
600        };
601        // Before the open, which can wait at the stream limit of the peer.
602        set_if_changed(&self.has_streams, true);
603        let mut stream = tokio::select! {
604            biased;
605            // Before the open, so that a revoked lease starts no header.
606            _ = revoked.cancelled() => return Ok(false),
607            open = n0_future::time::timeout(OPEN_TIMEOUT, self.conn.open_uni()) => {
608                open.map_err(|_| e!(WriteError::OpenStalled))??
609            }
610        };
611        StreamHeader::write(topic, &mut stream, &mut self.buffer, self.max_message_size).await?;
612        debug!(topic=%topic.fmt_short(), "stream opened");
613        // The QUIC stack of the peer acknowledges data that its app did not read
614        // yet, so this needs no timeout.
615        let (id, stopped) = (stream.id(), stream.stopped());
616        self.live_streams.push(Box::pin(async move {
617            stopped.await.ok();
618            (topic, id)
619        }));
620        entry.insert(stream);
621        Ok(true)
622    }
623}
