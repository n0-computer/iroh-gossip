13/// Remembers the good peers of a topic, and discovers peers for it.
14///
15/// Gossip calls [`Self::discover`] for a topic that has no neighbors, and
16/// joins the topic through each peer the stream yields. The hooks tell the
17/// implementation which peers proved good: a peer that became our neighbor,
18/// and a peer we received a message from. They are synchronous and called on
19/// the gossip actor, so they must be cheap. An implementation that needs to do
20/// IO in them should spawn it.
21///
22/// A peer we lose because we ourselves went offline looks failed to us, so an
23/// implementation that remembers peers should not forget them on failure.
24pub trait TopicDiscovery: std::fmt::Debug + Send + Sync + 'static {
25    /// We joined `topic`. For example, start announcing it.
26    fn topic_joined(&self, _topic: TopicId) {}
27
28    /// We left `topic`. Stop announcing it, and drop what was kept for it.
29    fn topic_left(&self, _topic: TopicId) {}
30
31    /// `peer` became our neighbor in `topic`.
32    fn neighbor_up(&self, _topic: TopicId, _peer: EndpointId) {}
33
34    /// We received a message in `topic` from `peer`, so it is reachable and in the topic.
35    ///
36    /// Called for every message, from neighbors and other peers alike, so it
37    /// must be very cheap. Use it to refresh a peer that stays a neighbor for
38    /// a long time.
39    fn peer_seen(&self, _topic: TopicId, _peer: EndpointId) {}
40
41    /// Returns a stream of peers to join `topic` through, for example the
42    /// results of a DHT lookup as they arrive.
43    ///
44    /// Called for a topic without neighbors, at most every 10 seconds, and
45    /// never while an earlier stream for the topic still runs. Gossip joins through each peer the stream yields, and drops
46    /// the stream once the topic has a neighbor. The stream is polled on the
47    /// gossip actor, so it must not block.
48    fn discover(&self, topic: TopicId) -> BoxStream<EndpointId>;
49}
50
51/// How often gossip checks for topics without neighbors, see [`TopicDiscovery::discover`].
52pub(crate) const DISCOVERY_INTERVAL: std::time::Duration = std::time::Duration::from_secs(10);
53
54/// Remembers the most recently seen good peers of each topic, in memory and up to a fixed number.
55///
56/// A peer counts as seen when it becomes our neighbor or sends us a message in
57/// the topic. Peers only leave by age, never because they failed, so a node
58/// that loses its own network can rejoin through the peers it had. All is
59/// forgotten when we leave the topic.
60#[derive(Debug)]
61pub struct RecentPeers {
62    capacity: usize,
63    /// Per topic, least recently seen first.
64    peers: Mutex<HashMap<TopicId, VecDeque<EndpointId>>>,
65}
66
67impl RecentPeers {
68    /// Returns a memory of up to `capacity` peers per topic.
69    pub fn new(capacity: usize) -> Self {
70        Self {
71            capacity,
72            peers: Default::default(),
73        }
74    }
75
76    fn touch(&self, topic: TopicId, peer: EndpointId) {
77        let mut peers = self.peers.lock().expect("poisoned");
78        let recent = peers.entry(topic).or_default();
79        recent.retain(|known| *known != peer);
80        recent.push_back(peer);
81        if recent.len() > self.capacity {
82            recent.pop_front();
83        }
84    }
85}
86
87impl Default for RecentPeers {
88    /// Remembers up to 8 peers per topic.
89    fn default() -> Self {
90        Self::new(8)
91    }
92}
93
94impl TopicDiscovery for RecentPeers {
95    fn topic_left(&self, topic: TopicId) {
96        self.peers.lock().expect("poisoned").remove(&topic);
97    }
98
99    fn neighbor_up(&self, topic: TopicId, peer: EndpointId) {
100        self.touch(topic, peer);
101    }
102
103    fn peer_seen(&self, topic: TopicId, peer: EndpointId) {
104        self.touch(topic, peer);
105    }
106
107    /// Yields the remembered peers, the most recently seen first.
108    fn discover(&self, topic: TopicId) -> BoxStream<EndpointId> {
109        let peers = self.peers.lock().expect("poisoned");
110        let recent: Vec<_> = peers
111            .get(&topic)
112            .into_iter()
113            .flatten()
114            .rev()
115            .copied()
116            .collect();
117        Box::pin(n0_future::stream::iter(recent))
118    }
119}

121#[cfg(test)]
122mod tests {
123    use iroh::SecretKey;
124    use n0_future::StreamExt;
125
126    use super::*;
127
128    fn peer(n: u8) -> EndpointId {
129        SecretKey::from_bytes(&[n; 32]).public()
130    }
131
132    fn discovered(recent: &RecentPeers, topic: TopicId) -> Vec<EndpointId> {
133        n0_future::future::block_on(recent.discover(topic).collect())
134    }
135
136    /// The most recently seen peers come first, and the oldest beyond the
137    /// capacity are dropped.
138    #[test]
139    fn recent_peers_keep_the_latest() {
140        let topic: TopicId = [1; 32].into();
141        let recent = RecentPeers::new(3);
142        recent.neighbor_up(topic, peer(1));
143        recent.neighbor_up(topic, peer(2));
144        recent.peer_seen(topic, peer(3));
145        recent.peer_seen(topic, peer(1));
146        recent.neighbor_up(topic, peer(4));
147        assert_eq!(discovered(&recent, topic), vec![peer(4), peer(1), peer(3)]);
148    }
149
150    /// Topics are remembered apart, and leaving one forgets it.
151    #[test]
152    fn recent_peers_are_per_topic() {
153        let a: TopicId = [1; 32].into();
154        let b: TopicId = [2; 32].into();
155        let recent = RecentPeers::default();
156        recent.neighbor_up(a, peer(1));
157        recent.neighbor_up(b, peer(2));
158        assert_eq!(discovered(&recent, a), vec![peer(1)]);
159        recent.topic_left(a);
160        assert!(discovered(&recent, a).is_empty());
161        assert_eq!(discovered(&recent, b), vec![peer(2)]);
162    }
