3use std::{fmt, hash::Hash};
4
5use bytes::Bytes;
6use serde::{de::DeserializeOwned, Deserialize, Serialize};

8/// The identifier for a peer.
9///
10/// The protocol implementation is generic over this trait. When implementing the protocol,
11/// a concrete type must be chosen that will then be used throughout the implementation to identify
12/// and index individual peers.
13///
14/// Note that the concrete type will be used in protocol messages. Therefore, implementations of
15/// the protocol are only compatible if the same concrete type is supplied for this trait.
16///
17/// It does not need to be a peer's address, as long as the networking layer can translate
18/// the value of its concrete type into an address.
19pub trait PeerIdentity: Hash + Eq + Ord + Copy + fmt::Debug + Serialize + DeserializeOwned {}
20impl<T> PeerIdentity for T where
21    T: Hash + Eq + Ord + Copy + fmt::Debug + Serialize + DeserializeOwned
22{
23}
24
25/// Opaque binary data that is transmitted on messages that introduce new peers.
26///
27/// Implementations may use these bytes to supply addresses or other information needed to connect
28/// to a peer that is not included in the peer's [`PeerIdentity`].
29#[derive(derive_more::Debug, Serialize, Deserialize, Clone, PartialEq, Eq, Default)]
30#[debug("PeerData({}b)", self.0.len())]
31pub struct PeerData(Bytes);
32
33impl PeerData {
34    /// Create a new [`PeerData`] from a byte buffer.
35    pub fn new(data: impl Into<Bytes>) -> Self {
36        Self(data.into())
37    }
38
39    /// Get the peer data as a byte slice.
40    pub fn as_bytes(&self) -> &[u8] {
41        &self.0
42    }
43}
