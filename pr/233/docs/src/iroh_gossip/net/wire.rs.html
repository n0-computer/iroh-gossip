1//! The gossip wire: frames, the stream header, and messages.
2//!
3//! All integers are LEB128 varints. A frame is a length followed by the
4//! payload. The first frame on a stream is the header, a kind followed by a
5//! list of parameters:
6//!
7//! ```text
8//! frame     = length:varint  payload
9//! header    = kind:varint  parameter*
10//! parameter = id:varint  length:varint  value:[u8; length]
11//! ```
12//!
13//! Every later frame is a message: a type code, then its fields in postcard.
14//! A reader skips a message of an unknown type and a frame over its size
15//! limit, and ignores bytes after the fields it knows.
16

31/// How long a write may make no progress before it fails the connection.
32///
33/// A peer extends its flow control window only after it read part of it, so
34/// the writes to a slow reader block for a while now and then.
35pub(super) const WRITE_TIMEOUT: Duration = Duration::from_secs(if cfg!(test) { 5 } else { 60 });
36
37/// How long opening a stream may wait at the stream limit of the peer before it fails the connection.
38///
39/// The peer raises its limit only once one of our streams ends. With as many
40/// topics as the limit, the wait does not end. The limit is the
41/// `max_concurrent_uni_streams` of the peer's endpoint, 100 by default. A
42/// stalled open of an initial topic only skips the other initial topics.
43pub(super) const OPEN_TIMEOUT: Duration = Duration::from_secs(if cfg!(test) { 5 } else { 15 });
44
45/// How long a peer may take to send a header, from the accept of its stream.
46///
47/// A header that is late fails the connection, as the streams after it wait for it.
48pub(super) const HEADER_TIMEOUT: Duration = Duration::from_secs(if cfg!(test) { 3 } else { 10 });
49
50/// The maximum length of a header frame payload, in bytes.
51///
52/// A longer header fails the connection.
53///
54/// Headers do not count against the message size limit.
55const MAX_HEADER_LEN: usize = 4096;

57/// Header kind of the control stream.
58const KIND_CONTROL: u64 = 0x00;

64///
65/// A longer one fails the connection.
66const MAX_CONTROL_FRAME: u64 = 32;

68/// Header kind of a stream of one topic.
69///
70/// The stream carries both the control and the data messages of the topic.
71const KIND_TOPIC: u64 = 0x01;
72
73/// Header parameter with the 32 byte topic id. Required on topic streams.
74const PARAM_TOPIC_ID: u64 = 0x00;

76/// The message type codes.
77mod types {
78    pub(super) const SWARM_JOIN: u64 = 0x01;
79    pub(super) const SWARM_FORWARD_JOIN: u64 = 0x02;
80    pub(super) const SWARM_SHUFFLE: u64 = 0x03;
81    pub(super) const SWARM_SHUFFLE_REPLY: u64 = 0x04;

84    pub(super) const SWARM_CLOSE: u64 = 0x07;

89    pub(super) const GOSSIP_IHAVE: u64 = 0x13;

95    /// Control stream: `opened: varint`, the number of topic streams the sender opened.
96    pub(super) const GOAWAY: u64 = 0x20;
97}

173/// Reads the payload of a header frame into `buffer`.
174///
175/// A header over [`MAX_HEADER_LEN`] fails before the reader reads its payload. The
176/// whole header must arrive in [`HEADER_TIMEOUT`].
177async fn read_header<R: AsyncRead + Unpin>(
178    reader: &mut R,
179    buffer: &mut Vec<u8>,
180) -> Result<(), ReadError> {
181    let read = async {
182        let len = read_prefix(reader)
183            .await?
184            .ok_or_else(|| io::Error::from(io::ErrorKind::UnexpectedEof))?;
185        if len > MAX_HEADER_LEN as u64 {
186            return Err(e!(ReadError::TooLarge));
187        }
188        read_payload(reader, buffer, len).await
189    };
190    n0_future::time::timeout(HEADER_TIMEOUT, read)
191        .await
192        .map_err(|_| io::Error::from(io::ErrorKind::TimedOut))?
193}

195/// Decodes the payload of a header frame of `kind` into its parameters, in order.
196///
197/// The decoder keeps a repeated parameter, and [`param`] finds the first.
198fn decode_header(mut buf: &[u8], kind: u64) -> Result<Vec<(u64, Bytes)>, ReadError> {
199    if take::<u64>(&mut buf)? != kind {
200        return Err(e!(ReadError::Malformed));
201    }
202    let mut params = Vec::new();
203    while !buf.is_empty() {
204        let id: u64 = take(&mut buf)?;
205        let value: &[u8] = take(&mut buf)?;
206        params.push((id, Bytes::copy_from_slice(value)));
207    }
208    Ok(params)
209}
210
211/// Returns the first parameter `id` of a header that has exactly `N` bytes.
212fn param<const N: usize>(params: &[(u64, Bytes)], id: u64) -> Result<[u8; N], ReadError> {
213    params
214        .iter()
215        .find(|(i, _)| *i == id)
216        .and_then(|(_, value)| value.as_ref().try_into().ok())
217        .ok_or_else(|| e!(ReadError::Malformed))
218}

220impl StreamHeader {
221    /// Reads the header frame.
222    pub(super) async fn read<R: AsyncRead + Unpin>(
223        reader: &mut R,
224        buffer: &mut Vec<u8>,
225    ) -> Result<Self, ReadError> {
226        read_header(reader, buffer).await?;
227        Self::decode(buffer)
228    }
229
230    /// Decodes the payload of a header frame.
231    fn decode(buf: &[u8]) -> Result<Self, ReadError> {
232        let params = decode_header(buf, KIND_TOPIC)?;
233        Ok(Self {
234            topic: TopicId::from_bytes(param(&params, PARAM_TOPIC_ID)?),
235        })
236    }
237
238    /// Writes the header that we send on a stream for `topic`.
239    pub(super) async fn write(
240        topic: TopicId,
241        stream: &mut SendStream,
242        buffer: &mut Vec<u8>,
243    ) -> Result<(), WriteError> {
244        let mut payload = Vec::new();
245        put_header(
246            &mut payload,
247            KIND_TOPIC,
248            &[(PARAM_TOPIC_ID, topic.as_bytes())],
249        );
250        buffer.clear();
251        put_varint(buffer, payload.len() as u64);
252        buffer.extend_from_slice(&payload);
253        write_all(stream, buffer).await
254    }
255}

257/// The code of a connection close.
258///
259/// The orderly closes come first, then the failures.
260#[derive(Debug, Clone, Copy, PartialEq, Eq)]
261pub(super) enum CloseCode {
262    /// A finished retire.
263    ///
264    /// This is not 0. noq sends code 0 when the last handle of a connection drops without a
265    /// close, as in a crash or a dropped `Gossip`. Such a close must not read as a finished
266    /// retire.
267    Retired = 1,
268    /// The node does not need the connection: it lost a crossed dial, or the peer restarted.
269    Stale = 2,
270    /// The node shuts down.
271    Shutdown = 3,
272    /// The peer broke the protocol.
273    Protocol = 4,
274    /// A write to the peer failed.
275    SendFailed = 5,
276    /// A queue, a write, a stream open, or a retire made no progress in time.
277    Blocked = 6,
278}
279
280impl CloseCode {
281    /// Returns the code of a close, or `None` for an unknown one.
282    pub(super) fn from_code(code: VarInt) -> Option<Self> {
283        Some(match code.into_inner() {
284            1 => Self::Retired,
285            2 => Self::Stale,
286            3 => Self::Shutdown,
287            4 => Self::Protocol,
288            5 => Self::SendFailed,
289            6 => Self::Blocked,
290            _ => return None,
291        })
292    }
293
294    pub(super) fn code(self) -> VarInt {
295        VarInt::from_u32(self as u32)
296    }
297}

299/// A random id that a node draws once at start, and sends in the header of each control stream.
300///
301/// The control stream is the first bidirectional stream of a connection. The
302/// dialer opens it and writes its header. The acceptor answers with its own.
303#[derive(Clone, Copy, PartialEq, Eq, Hash, derive_more::Debug)]
304#[debug("{}", hex::encode(_0))]
305pub(super) struct Session(pub(super) [u8; 8]);
306
307impl Session {
308    pub(super) fn random() -> Self {
309        Self(rand::random())
310    }
311
312    /// Returns the frame of the control header.
313    pub(super) fn to_frame(self) -> Vec<u8> {
314        let mut payload = Vec::new();
315        put_header(&mut payload, KIND_CONTROL, &[(PARAM_SESSION, &self.0)]);
316        framed(&payload)
317    }
318
319    pub(super) async fn read<R: AsyncRead + Unpin>(
320        reader: &mut R,
321        buffer: &mut Vec<u8>,
322    ) -> Result<Self, ReadError> {
323        read_header(reader, buffer).await?;
324        Self::decode(buffer)
325    }
326
327    fn decode(buf: &[u8]) -> Result<Self, ReadError> {
328        Ok(Self(param(
329            &decode_header(buf, KIND_CONTROL)?,
330            PARAM_SESSION,
331        )?))
332    }
333}

335/// A message on the control stream after the headers.
336#[derive(Debug, Clone, Copy, PartialEq, Eq)]
337pub(super) enum ControlMessage {
338    /// The writer of the sender stopped, after it opened this number of topic streams.
339    Goaway(u64),
340}
341
342impl ControlMessage {
343    /// Returns the frame of the message.
344    pub(super) fn to_frame(self) -> Vec<u8> {
345        let Self::Goaway(opened) = self;
346        let mut payload = Vec::new();
347        put_varint(&mut payload, types::GOAWAY);
348        put_varint(&mut payload, opened);
349        framed(&payload)
350    }
351
352    /// Reads the next message, and skips messages of an unknown type.
353    ///
354    /// The end of the stream is an error, as is a frame over [`MAX_CONTROL_FRAME`].
355    pub(super) async fn read<R: AsyncRead + Unpin>(
356        reader: &mut R,
357        buffer: &mut Vec<u8>,
358    ) -> Result<Self, ReadError> {
359        loop {
360            let len = read_prefix(reader)
361                .await?
362                .ok_or_else(|| io::Error::from(io::ErrorKind::UnexpectedEof))?;
363            if len > MAX_CONTROL_FRAME {
364                return Err(e!(ReadError::TooLarge));
365            }
366            read_payload(reader, buffer, len).await?;
367            if let Some(message) = Self::decode(buffer)? {
368                return Ok(message);
369            }
370        }
371    }
372
373    fn decode(mut buf: &[u8]) -> Result<Option<Self>, ReadError> {
374        let buf = &mut buf;
375        Ok(Some(match take(buf)? {
376            types::GOAWAY => Self::Goaway(take(buf)?),
377            _ => return Ok(None),
378        }))
379    }
380}

382/// Returns `payload` as a frame.
383fn framed(payload: &[u8]) -> Vec<u8> {
384    let mut buf = Vec::new();
385    put_varint(&mut buf, payload.len() as u64);
386    buf.extend_from_slice(payload);
387    buf
388}
389
390/// Appends the payload of a header frame.
391fn put_header(buf: &mut Vec<u8>, kind: u64, params: &[(u64, &[u8])]) {
392    put_varint(buf, kind);
393    for (id, value) in params {
394        put_varint(buf, *id);
395        put_varint(buf, value.len() as u64);
396        buf.extend_from_slice(value);
397    }
398}

400/// Reads the next message of a stream, or `None` at its end.
401///
402/// Skips frames over `max_message_size` without reading them into memory,
403/// and messages of an unknown type.
404pub(super) async fn read_message<R: AsyncRead + Unpin>(
405    reader: &mut R,
406    buffer: &mut Vec<u8>,
407    max_message_size: usize,
408) -> Result<Option<WireMessage>, ReadError> {
409    while let Some(len) = read_prefix(reader).await? {
410        if len > max_message_size as u64 {
411            debug!(
412                len,
413                max = max_message_size,
414                "skip a frame over the size limit"
415            );
416            let skipped =
417                tokio::io::copy(&mut (&mut *reader).take(len), &mut tokio::io::sink()).await?;
418            if skipped < len {
419                return Err(io::Error::from(io::ErrorKind::UnexpectedEof).into());
420            }
421            continue;
422        }
423        read_payload(reader, buffer, len).await?;
424        if let Some(message) = decode_message(buffer)? {
425            return Ok(Some(message));
426        }
427    }
428    Ok(None)
429}

431/// Reads the length prefix of a frame, or `None` at the end of the stream.
432async fn read_prefix<R: AsyncRead + Unpin>(reader: &mut R) -> Result<Option<u64>, ReadError> {
433    let mut len = 0u64;
434    for i in 0..MAX_VARINT_LEN {
435        let byte = match reader.read_u8().await {
436            Ok(byte) => byte,
437            Err(err) if err.kind() == io::ErrorKind::UnexpectedEof && i == 0 => return Ok(None),
438            Err(err) => return Err(err.into()),
439        };
440        // The tenth byte holds the last bit of a `u64`.
441        len |= u64::from(byte & 0x7f)
442            .checked_mul(1 << (7 * i))
443            .ok_or_else(|| e!(ReadError::TooLarge))?;
444        if byte & 0x80 == 0 {
445            return Ok(Some(len));
446        }
447    }
448    Err(e!(ReadError::TooLarge))
449}

451/// Reads a payload of `len` bytes into the buffer.
452///
453/// The buffer reserves at most 64 KiB up front, and grows past that only as the bytes arrive.
454async fn read_payload<R: AsyncRead + Unpin>(
455    reader: &mut R,
456    buffer: &mut Vec<u8>,
457    len: u64,
458) -> Result<(), ReadError> {
459    buffer.clear();
460    buffer.reserve(len.min(64 * 1024) as usize);
461    let read = reader.take(len).read_to_end(buffer).await?;
462    if (read as u64) < len {
463        return Err(io::Error::from(io::ErrorKind::UnexpectedEof).into());
464    }
465    Ok(())
466}

468/// A message as it goes on the wire: its type code, then its fields.
469///
470/// The fields are the postcard encoding of the proto type.
471struct WireBody<'a>(&'a WireMessage);
472
473impl Serialize for WireBody<'_> {
474    fn serialize<S: serde::Serializer>(&self, s: S) -> Result<S::Ok, S::Error> {
475        use hyparview::Message as Swarm;
476        use plumtree::Message as Broadcast;
477        match self.0 {
478            topic::Message::Swarm(message) => match message {
479                Swarm::Join(data) => (types::SWARM_JOIN, data).serialize(s),
480                Swarm::ForwardJoin(join) => (types::SWARM_FORWARD_JOIN, join).serialize(s),
481                Swarm::Shuffle(shuffle) => (types::SWARM_SHUFFLE, shuffle).serialize(s),
482                Swarm::ShuffleReply(reply) => (types::SWARM_SHUFFLE_REPLY, reply).serialize(s),
483                Swarm::NeighborRequest(request) => {
484                    (types::SWARM_NEIGHBOR_REQUEST, request).serialize(s)
485                }
486                Swarm::NeighborReply(data) => (types::SWARM_NEIGHBOR_REPLY, data).serialize(s),
487                Swarm::Close(close) => (types::SWARM_CLOSE, close).serialize(s),
488            },
489            topic::Message::Gossip(message) => match message {
490                Broadcast::Gossip(gossip) => match gossip.scope {
491                    DeliveryScope::Swarm(round) => {
492                        (types::GOSSIP_MESSAGE, &gossip.content, round).serialize(s)
493                    }
494                    DeliveryScope::Neighbors => {
495                        (types::GOSSIP_MESSAGE_NEIGHBOR, &gossip.content).serialize(s)
496                    }
497                },
498                Broadcast::Prune => (types::GOSSIP_PRUNE,).serialize(s),
499                Broadcast::Graft(graft) => (types::GOSSIP_GRAFT, graft).serialize(s),
500                Broadcast::IHave(ihave) => (types::GOSSIP_IHAVE, ihave).serialize(s),
501            },
502        }
503    }
504}

506/// Returns how many bytes `message` takes encoded, without the frame prefix.
507pub(super) fn message_len(message: &WireMessage) -> Result<usize, postcard::Error> {
508    postcard::experimental::serialized_size(&WireBody(message))
509}

511/// Decodes a message, and ignores the bytes after its fields.
512///
513/// The body is the postcard encoding of the proto type. Fields that a later
514/// version appends sit in the bytes after the known fields.
515///
516/// Returns `None` for a message of an unknown type, which the reader skips.
517fn decode_message(mut buf: &[u8]) -> Result<Option<WireMessage>, ReadError> {
518    use hyparview::Message as Swarm;
519    use plumtree::Message as Broadcast;
520    let buf = &mut buf;
521    let message = match take(buf)? {
522        types::SWARM_JOIN => WireMessage::Swarm(Swarm::Join(take(buf)?)),
523        types::SWARM_FORWARD_JOIN => WireMessage::Swarm(Swarm::ForwardJoin(take(buf)?)),
524        types::SWARM_SHUFFLE => WireMessage::Swarm(Swarm::Shuffle(take(buf)?)),
525        types::SWARM_SHUFFLE_REPLY => WireMessage::Swarm(Swarm::ShuffleReply(take(buf)?)),
526        types::SWARM_NEIGHBOR_REQUEST => WireMessage::Swarm(Swarm::NeighborRequest(take(buf)?)),
527        types::SWARM_NEIGHBOR_REPLY => WireMessage::Swarm(Swarm::NeighborReply(take(buf)?)),
528        types::SWARM_CLOSE => WireMessage::Swarm(Swarm::Close(take(buf)?)),
529        types::GOSSIP_MESSAGE => {
530            let (content, round): (Bytes, Round) = take(buf)?;
531            let scope = DeliveryScope::Swarm(round);
532            WireMessage::Gossip(Broadcast::Gossip(plumtree::Gossip::new(content, scope)))
533        }
534        types::GOSSIP_MESSAGE_NEIGHBOR => {
535            let content: Bytes = take(buf)?;
536            let scope = DeliveryScope::Neighbors;
537            WireMessage::Gossip(Broadcast::Gossip(plumtree::Gossip::new(content, scope)))
538        }
539        types::GOSSIP_PRUNE => WireMessage::Gossip(Broadcast::Prune),
540        types::GOSSIP_GRAFT => WireMessage::Gossip(Broadcast::Graft(take(buf)?)),
541        types::GOSSIP_IHAVE => WireMessage::Gossip(Broadcast::IHave(take(buf)?)),
542        _ => return Ok(None),
543    };
544    Ok(Some(message))
545}

547/// Reads a value from the front of `buf`, and moves `buf` past it.
548///
549/// Integers are LEB128 varints, as in postcard.
550fn take<'a, T: Deserialize<'a>>(buf: &mut &'a [u8]) -> Result<T, postcard::Error> {
551    let (value, rest) = postcard::take_from_bytes(buf)?;
552    *buf = rest;
553    Ok(value)
554}
555
556/// Appends `value` as a LEB128 varint.
557fn put_varint(buf: &mut Vec<u8>, value: u64) {
558    *buf = postcard::to_extend(&value, std::mem::take(buf)).expect("a varint encodes");
559}

561/// Writes a message as a frame, with `len` as its encoded size from [`message_len`].
562pub(super) async fn write_sized_message(
563    stream: &mut SendStream,
564    message: &WireMessage,
565    len: usize,
566    buffer: &mut Vec<u8>,
567    max_message_size: usize,
568) -> Result<(), WriteError> {
569    if len > max_message_size {
570        return Err(e!(WriteError::TooLarge));
571    }
572    buffer.clear();
573    put_varint(buffer, len as u64);
574    let start = buffer.len();
575    buffer.resize(start + len, 0u8);
576    let encoded = postcard::to_slice(&WireBody(message), &mut buffer[start..])?.len();
577    debug_assert_eq!(encoded, len, "the length prefix does not match");
578    write_all(stream, buffer).await
579}

593/// Writes all of `buffer`.
594pub(super) async fn write_all(stream: &mut SendStream, buffer: &[u8]) -> Result<(), WriteError> {
595    // Each write takes some bytes, or fails once the peer takes none for long.
596    // A slow peer thus keeps the connection, however long the frame takes.
597    let mut sent = 0;
598    while sent < buffer.len() {
599        sent += n0_future::time::timeout(WRITE_TIMEOUT, stream.write(&buffer[sent..]))
600            .await
601            .map_err(|_| e!(WriteError::Stalled))?
602            .map_err(|err| match err {
603                iroh::endpoint::WriteError::Stopped(code) => e!(WriteError::Stopped {
604                    code: code.into_inner()
605                }),
606                other => WriteError::from(io::Error::other(other)),
607            })?;
608    }
609    Ok(())
610}

621    fn header(params: &[(u64, &[u8])]) -> Vec<u8> {

627    fn encode(message: &WireMessage) -> Vec<u8> {
628        let mut buf = vec![0; message_len(message).unwrap()];
629        let len = postcard::to_slice(&WireBody(message), &mut buf)
630            .unwrap()
631            .len();
632        assert_eq!(len, buf.len());
633        buf
634    }

693        let vectors = vectors();
694        let kinds: std::collections::BTreeSet<_> = vectors.iter().map(|v| &v[..2]).collect();
695        assert_eq!(kinds.len(), 12, "a type has no vector");
696        for vector in vectors {

698            let bytes = encode(&message);

770    /// A header over the limit fails on its length, before its payload.
771    #[tokio::test]
772    async fn oversized_header_is_rejected() {
773        // The kind and the topic parameter take 35 bytes. The id and length of the filler
774        // take 3 bytes.
775        let frame = |filler: usize| {
776            let filler = vec![0; filler];
777            framed(&header(&[(PARAM_TOPIC_ID, &TOPIC[..]), (58, &filler[..])]))
778        };
779        let fits = frame(MAX_HEADER_LEN - 38);
780        assert_eq!(fits.len(), 2 + MAX_HEADER_LEN);
781        let ok = StreamHeader::read(&mut &fits[..], &mut Vec::new()).await;
782        assert!(ok.is_ok());
783        // Only the prefix is there, so the reader did not wait for the payload.
784        let mut over = frame(MAX_HEADER_LEN - 37);
785        over.truncate(2);
786        let mut buffer = Vec::new();
787        let err = StreamHeader::read(&mut &over[..], &mut buffer)
788            .await
789            .unwrap_err();
790        assert!(matches!(err, ReadError::TooLarge { .. }), "{err}");
791        assert_eq!(buffer.capacity(), 0, "the reader allocated");
792    }

884    /// `max_gossip_size` is the size at the largest round, and no less at any other.
885    #[test]
886    fn max_gossip_size_covers_every_round() {
887        use crate::proto::Scope;
888        let size = |content: &Bytes, scope| {
889            let gossip = plumtree::Gossip::new(content.clone(), scope);
890            message_len(&WireMessage::Gossip(plumtree::Message::Gossip(gossip))).unwrap()
891        };
892        for len in [0, 127, 128, 4000, 20_000] {
893            let content = Bytes::from(vec![0; len]);
894            let max = plumtree::max_gossip_size(len, Scope::Swarm);
895            for round in [0, 127, 128, 16_383, 16_384] {
896                assert!(size(&content, DeliveryScope::Swarm(Round::from(round))) <= max);
897            }
898            assert_eq!(
899                size(&content, DeliveryScope::Swarm(Round::from(u16::MAX))),
900                max
901            );
902            assert_eq!(
903                size(&content, DeliveryScope::Neighbors),
904                plumtree::max_gossip_size(len, Scope::Neighbors)
905            );
906        }
907    }
