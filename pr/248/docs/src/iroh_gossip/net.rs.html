48/// Publish and subscribe on gossiping topics.
49///
50/// Each topic is a separate broadcast tree with separate memberships.
51///
52/// A topic has to be joined before you can publish or subscribe on the topic.
53/// To join the swarm for a topic, you have to know the [`PublicKey`] of at least one peer that also joined the topic.
54///
55/// Messages published on the swarm will be delivered to all peers that joined the swarm for that
56/// topic. You will also be relaying (gossiping) messages published by other peers.
57///
58/// With the default settings, a node has up to 5 active neighbors per topic (the default active view).
59///
60/// Even though the [`Gossip`] is created from a [`Endpoint`], it does not accept connections
61/// itself. You should run an accept loop on the [`Endpoint`] yourself, check the ALPN protocol of incoming
62/// connections, and if the ALPN protocol equals [`ALPN`], forward the connection to the
63/// gossip actor through [Self::handle_connection].
64///
65/// The gossip actor will, however, initiate new connections to other peers by itself.
66/// It closes a connection ten seconds after neither side has a stream on it. This includes
67/// connections that you passed to [`Self::handle_connection`], so do not use them for other traffic.
68///
69/// The actor stops when you drop the last clone of [`Gossip`]. Keep a clone for as long as you
70/// need the topics. Call [`Self::shutdown`] to leave the topics cleanly first.
71#[derive(Debug, Clone)]
72pub struct Gossip {
73    inner: Arc<Inner>,
74}
75
76/// A message from the [`Gossip`] handle to the actor.
77#[derive(Debug)]
78enum ToActor {
79    Join(Join),
80    Connection(Connection),
81    Shutdown { reply: oneshot::Sender<()> },
82}
83
84/// A request from [`Gossip::subscribe_with_opts`] to the actor to join a topic.
85#[derive(Debug)]
86struct Join {
87    topic_id: TopicId,
88    bootstrap: BTreeSet<EndpointId>,
89    events_tx: mpsc::Sender<Event>,
90    commands_rx: mpsc::Receiver<Command>,
91    /// The actor answers once it registered the topic and applied the join.
92    registered: oneshot::Sender<()>,
93}
94
95#[derive(Debug)]
96struct Inner {
97    to_actor: mpsc::Sender<ToActor>,
98    _actor_handle: AbortOnDropHandle<()>,
99    max_message_size: usize,
100    /// The largest content that a broadcast takes.
101    max_content: usize,
102    metrics: Arc<Metrics>,
103}
104
105impl ProtocolHandler for Gossip {
106    async fn accept(&self, connection: Connection) -> Result<(), AcceptError> {
107        self.handle_connection(connection)
108            .await
109            .map_err(AcceptError::from_err)?;
110        Ok(())
111    }
112
113    async fn shutdown(&self) {
114        if let Err(err) = self.shutdown().await {
115            warn!("error while shutting down gossip: {err:#}");
116        }
117    }
118}

138        self.config.proto.max_message_size = size;
139        self
140    }

180    /// Spawns a gossip actor, and returns its handle.
181    ///
182    /// This has to run inside a tokio runtime. The actor stops when you drop the last clone of
183    /// the returned [`Gossip`].
184    pub fn spawn(self, endpoint: Endpoint) -> Gossip {
185        let metrics = Arc::new(Metrics::default());
186        let address_lookup = GossipAddressLookup::default();
187
188        // `Endpoint::address_lookup` returns `Err` when the endpoint is closed.
189        // In that case, the gossip actor will close too very soon for other reasons,
190        // so it's fine if we only add our `GossipAddressLookup` for the non-closed
191        // case. The alternative would be to return a `Result` from `spawn`,
192        // but as long as this is the only direct error case, it seems unwarranted.
193        if let Ok(endpoint_addr_lookup) = endpoint.address_lookup().as_ref() {
194            endpoint_addr_lookup.add(address_lookup.clone());
195        }
196        let me = endpoint.id().fmt_short();
197        let max_message_size = self.config.proto.max_message_size;
198        let (actor, to_actor) = Actor::new(endpoint, self.config, metrics.clone(), address_lookup);
199
200        let actor_handle = task::spawn(actor.run().instrument(error_span!("gossip", %me)));
201
202        Gossip::new(
203            to_actor,
204            AbortOnDropHandle::new(actor_handle),
205            max_message_size,
206            metrics,
207        )
208    }
209}

211/// Options of the send queue to each peer, for [`Builder::send_queue`].
212///
213/// Gossip keeps one send queue per peer, shared between all topics. The queue
214/// holds up to [`Self::size`] bytes of messages on their way to that peer.
215/// When the queue becomes full, new data and announcement messages to that peer
216/// will be dropped to not hold up the whole swarm. If this peer cannot keep
217/// up with the volume of messages to be sent or relayed, the following
218/// may happen:
219///
220/// - A broadcast to the peer is dropped, and [`Metrics::msgs_dropped`] counts
221///   it. Gossip announces the broadcast to the peer again, and the peer
222///   fetches it while it is still cached, 30 s by default. A broadcast to
223///   direct neighbors is never repaired.
224/// - A control message to the peer closes the connection, and
225///   [`Metrics::peers_send_queue_full`] counts it. Both sides find new
226///   neighbors.
227/// - Once the queues to all peers that a topic pushes to are half full of data,
228///   [`GossipSender::broadcast`] on that topic waits for room, up to 100 ms
229///   per message. Relayed traffic and other topics fill the same queues, so
230///   this slows a publisher that sends little as well.
231///
232/// Independent of the size:
233///
234/// - Messages to a slow peer arrive late. The queue bounds bytes, not time,
235///   and QUIC buffers more on top.
236/// - A peer that takes no data for 60 s loses its connection, as does one
237///   that shares more topics with this node than its stream limit allows,
238///   100 by default.
239///
240/// [`GossipSender::broadcast`]: crate::GossipSender::broadcast
241#[derive(Debug, Clone, Copy)]
242pub struct QueueOptions {
243    size: usize,
244}
245
246impl Default for QueueOptions {
247    fn default() -> Self {
248        Self {
249            size: queue::QUEUE_SIZE,
250        }
251    }
252}
253
254impl QueueOptions {
255    /// Sets the size of the queue in bytes, 1 MiB by default.
256    ///
257    /// A larger queue absorbs larger bursts and longer stalls without a
258    /// loss. It costs up to its size in memory for each peer, and it lets
259    /// more old data wait for a slow one. A message takes its encoded size
260    /// and about 130 bytes more. Whatever the setting, the queue is at
261    /// least 64 KiB, and fits eight messages of [`Builder::max_message_size`].
262    pub fn size(mut self, size: usize) -> Self {
263        self.size = size;
264        self
265    }
266}

295    /// Joins a gossip topic with options.
296    ///
297    /// Returns a [`GossipTopic`] once the node has registered the topic, so it accepts a join
298    /// from a peer from then on. It does not wait for a neighbor. To wait for the first
299    /// neighbor, await [`GossipTopic::joined`].
300    ///
301    /// See [`JoinOptions`] for available options.
302    ///
303    /// A broadcast reaches only the neighbors that exist when the node sends it. Await
304    /// [`GossipTopic::joined`] before the first broadcast. With no bootstrap peers, the node
305    /// waits until another node joins it.
306    ///
307    /// The node leaves the topic when you drop all handles of the topic.
308    ///
309    /// A [`JoinOptions::subscription_capacity`] of 0 counts as 1.
310    ///
311    /// # Errors
312    ///
313    /// Returns [`Error::Closed`] if the gossip actor stopped.
314    pub async fn subscribe_with_opts(
315        &self,
316        topic_id: TopicId,
317        opts: JoinOptions,
318    ) -> Result<GossipTopic, Error> {
319        let (commands_tx, commands_rx) = mpsc::channel(TOPIC_COMMANDS_CAP);
320        let (events_tx, events_rx) = mpsc::channel(opts.subscription_capacity.max(1));
321        let (registered_tx, registered_rx) = oneshot::channel();
322        self.inner
323            .to_actor
324            .send(ToActor::Join(Join {
325                topic_id,
326                bootstrap: opts.bootstrap,
327                events_tx,
328                commands_rx,
329                registered: registered_tx,
330            }))
331            .await
332            .map_err(|_| e!(Error::Closed))?;
333        registered_rx.await.map_err(|_| e!(Error::Closed))?;
334        Ok(GossipTopic::new(
335            commands_tx,
336            events_rx,
337            self.inner.max_content,
338        ))
339    }

341    /// Joins a gossip topic with the default options, and waits for the first neighbor.
342    ///
343    /// This call has no timeout. If no peer becomes a neighbor, it waits forever. This happens
344    /// when all bootstrap peers are offline. Wrap the call in a timeout to stop the wait.
345    pub async fn subscribe_and_join(
346        &self,
347        topic_id: TopicId,
348        bootstrap: Vec<EndpointId>,
349    ) -> Result<GossipTopic, Error> {
350        let mut sub = self
351            .subscribe_with_opts(topic_id, JoinOptions::with_bootstrap(bootstrap))
352            .await?;
353        sub.joined().await?;
354        Ok(sub)
355    }
356
357    /// Joins a gossip topic with the default options.
358    ///
359    /// It does not wait for a bootstrap peer. To wait for the first neighbor, use
360    /// [`GossipTopic::joined`] or [`Self::subscribe_and_join`].
361    ///
362    /// # Errors
363    ///
364    /// Returns [`Error::Closed`] if the gossip actor stopped.
365    pub async fn subscribe(
366        &self,
367        topic_id: TopicId,
368        bootstrap: Vec<EndpointId>,
369    ) -> Result<GossipTopic, Error> {
370        self.subscribe_with_opts(topic_id, JoinOptions::with_bootstrap(bootstrap))
371            .await
372    }

384    /// Handles an incoming [`Connection`].
385    ///
386    /// The connection should be an accepted, server-side connection.
387    /// Make sure to check the ALPN protocol yourself before passing the connection.
388    ///
389    /// Gossip owns the connection from now on. It closes the connection when neither side has
390    /// had a stream on it for ten seconds.
391    ///
392    /// # Errors
393    ///
394    /// Returns [`Error::Closed`] if the gossip actor stopped.
395    pub async fn handle_connection(&self, conn: Connection) -> Result<(), Error> {
396        self.inner
397            .to_actor
398            .send(ToActor::Connection(conn))
399            .await
400            .map_err(|_| e!(Error::Closed))?;
401        Ok(())
402    }
403
404    /// Shutdown the gossip instance.
405    ///
406    /// This leaves all topics, sending `Disconnect` messages to peers, and then
407    /// stops the gossip actor loop and drops all state and connections. A gossip
408    /// instance that is shut down already, or shutting down, returns `Ok` too.
409    pub async fn shutdown(&self) -> Result<(), Error> {
410        let (reply, reply_rx) = oneshot::channel();
411        let shutdown = ToActor::Shutdown { reply };
412        // The actor is gone or about to go, which is what the caller asks for.
413        if self.inner.to_actor.send(shutdown).await.is_ok() {
414            reply_rx.await.ok();
415        }
416        Ok(())
417    }
418
419    /// Returns the metrics tracked for this gossip instance.
420    pub fn metrics(&self) -> &Arc<Metrics> {
421        &self.inner.metrics
422    }
