203/// Identifies the broadcast tree a message travels on.
204///
205/// A sender uses the shared tree `TreeId::default()` for its first
206/// [`Config::source_tree_after`] broadcasts. From the next one on, it uses a tree of
207/// its own, named with [`Self::of`]. A duplicate of one tree's message only
208/// prunes that tree. The origin names the tree, and every message of it carries
209/// the id: receivers copy it and never compute it. Two origins whose ids collide
210/// share a tree, which costs latency but not delivery.
211#[derive(Serialize, Deserialize, Clone, Copy, Debug, Default, PartialEq, Eq, Hash, MaxSize)]
212pub struct TreeId([u8; 8]);
213
214impl TreeId {
215    /// Returns the id of the tree that `origin` broadcasts on.
216    ///
217    /// The id is the first 8 bytes of the BLAKE3 hash of the encoded `origin`. An endpoint
218    /// id encodes as its 32 bytes.
219    pub fn of<PI: Serialize>(origin: &PI) -> Self {
220        let encoded = postcard::to_stdvec(origin).expect("a peer id serializes");
221        let hash = blake3::hash(&encoded);
222        Self(hash.as_bytes()[..8].try_into().expect("8 of 32 bytes"))
223    }
224}

304/// Returns the largest wire size of a [`Message::Gossip`] with `content_len` bytes of content.
305///
306/// A broadcast to the swarm is a type code, the content with its length prefix,
307/// the round and the tree. Each hop adds one to the round, so the encoding grows on the
308/// way. This assumes the largest round. A broadcast to neighbors has no round.
309pub(crate) fn max_gossip_size(content_len: usize, scope: Scope) -> usize {
310    let type_size = 1;
311    let prefix_size = (usize::BITS - content_len.leading_zeros())
312        .div_ceil(7)
313        .max(1) as usize;
314    let round_size = match scope {
315        // A varint of `u16::MAX`, and the tree.
316        Scope::Swarm => 3 + size_of::<TreeId>(),
317        Scope::Neighbors => 0,
318    };
319    type_size + prefix_size + content_len + round_size
320}

368/// Configuration for the gossip broadcast layer.
369///
370/// Currently, the expectation is that the configuration is the same for all peers in the
371/// network (as recommended in the paper).
372#[derive(Clone, Debug, Serialize, Deserialize)]
373#[serde(default)]
374#[non_exhaustive]
375pub struct Config {
376    /// When receiving an `IHave` message, this timeout is registered. If the message for the
377    /// `IHave` was not received once the timeout is expired, a `Graft` message is sent to the
378    /// peer that sent us the `IHave` to request the message payload.
379    ///
380    /// The plumtree paper notes:
381    /// > The timeout value is a protocol parameter that should be configured considering the
382    /// > diameter of the overlay and a target maximum recovery latency, defined by the application
383    /// > requirements. (p.8)
384    pub graft_timeout_1: Duration,
385    /// This timeout is registered when sending a `Graft` message. If a reply has not been
386    /// received once the timeout expires, we send another `Graft` message to the next peer that
387    /// sent us an `IHave` for this message.
388    ///
389    /// The plumtree paper notes:
390    /// > This second timeout value should be smaller that the first, in the order of an average
391    /// > round trip time to a neighbor.
392    pub graft_timeout_2: Duration,
393    /// Timeout after which `IHave` messages are pushed to peers.
394    pub dispatch_timeout: Duration,
395    /// The protocol performs a tree optimization, which promotes lazy peers to eager peers if the
396    /// `Message::IHave` messages received from them have a lower number of hops from the
397    /// message's origin as the `InEvent::Broadcast` messages received from our eager peers. This
398    /// parameter is the number of hops that the lazy peers must be closer to the origin than our
399    /// eager peers to be promoted to become an eager peer.
400    pub optimization_threshold: Round,
401
402    /// Duration for which to keep gossip messages in the internal message cache.
403    ///
404    /// Messages broadcast from this node or received from other nodes are kept in an internal
405    /// cache for this duration before being evicted. If this is too low, other nodes will not be
406    /// able to retrieve messages once they need them. If this is high, the cache will grow.
407    ///
408    /// Should be at least around several round trip times to peers.
409    ///
410    /// The cache, the message ids and the verdicts of neighbor messages are bounded by time
411    /// only, not by count. Their size grows with the rate at which peers send messages, times
412    /// the retention. A peer cannot send faster than its connection and the send queues of the
413    /// swarm allow, and a message filter can drop what a peer sends before it enters the cache.
414    /// A node that takes data from peers it does not trust needs a filter, or a shorter retention.
415    pub message_cache_retention: Duration,
416
417    /// Duration for which to keep the `MessageId`s for received messages.
418    ///
419    /// Should be at least as long as [`Self::message_cache_retention`], usually will be longer to
420    /// not accidentally receive messages multiple times.
421    pub message_id_retention: Duration,
422
423    /// How often the internal caches will be checked for expired items.
424    pub cache_evict_interval: Duration,
425
426    /// How many of our own broadcasts go on the shared tree before we get a tree of our own.
427    ///
428    /// Our first `source_tree_after` broadcasts name the shared tree [`TreeId::default()`], which
429    /// every node uses. From the next one on we name `TreeId::of(me)`. A node that first sees
430    /// that tree starts it as a copy of its shared tree, so the switch costs no flood. The tree
431    /// then diverges through the prunes and grafts of our traffic, like the paper's tree per
432    /// sender. The count starts again after we have not broadcast for five minutes.
433    ///
434    /// `0` gives every sender its own tree from its first message, which has to flood because
435    /// there is no shared tree to copy yet. A very high value gives one shared tree.
436    pub source_tree_after: u32,
437
438    /// How many requests a node in pull mode keeps in flight.
439    ///
440    /// A node with a link slower than the broadcast cannot take the eager pushes. They wait in
441    /// queues in front of its link, and each `Graft` for a message that waits there adds a
442    /// second copy. Such a node enters pull mode: its neighbors hold it lazy in every tree, and
443    /// it requests each announced message itself, oldest first, with at most this many requests
444    /// in flight. Its link then carries only the copies it asked for.
445    ///
446    /// A node enters pull mode when 16 messages did not arrive 500 ms after their announcement.
447    /// It leaves after 3 s with at most this many such messages. `0` turns pull mode off.
448    pub pull_window: usize,
449}

451impl Default for Config {
452    /// Returns the default plumtree configuration.
453    ///
454    /// The paper gives no numbers for the timeouts, so we chose them for typical
455    /// latencies of tens of milliseconds.
456    fn default() -> Self {
457        Self {
458            // Paper: "The timeout value is a protocol parameter that should be configured considering
459            // the diameter of the overlay and a target maximum recovery latency, defined by the
460            // application requirements. This is a parameter that should be statically configured
461            // at deployment time." (p. 8)
462            // In the paper it is a few roundtrips.
463            graft_timeout_1: Duration::from_millis(80),
464
465            // Paper: "This second timeout value should be smaller that the first, in the order of an
466            // average round trip time to a neighbor." (p. 9)
467            graft_timeout_2: Duration::from_millis(40),
468
469            // Again, paper does not tell a recommended number here. Likely should be quite small,
470            // as to not delay messages without need. This would also be the time frame in which
471            // `IHave`s are aggregated to save on packets.
472            dispatch_timeout: Duration::from_millis(5),
473
474            // This number comes from experiment settings the plumtree paper (p. 12)
475            optimization_threshold: Round(7),
476
477            // This is a certainly-high-enough value for usual operation.
478            message_cache_retention: Duration::from_secs(30),
479            message_id_retention: Duration::from_secs(90),
480            cache_evict_interval: Duration::from_secs(1),
481
482            // The first message builds the shared tree, and a sender's second one already
483            // names its own tree, copied from the shared one. Measured in the simulator: lower
484            // redundancy than 3 when all nodes send, equal when few do. See `TreeId`.
485            source_tree_after: 1,
486
487            // Measured with one node on a link at half the broadcast rate: windows of 4, 16 and
488            // 64 gave the same unique deliveries within the spread, and 4 the fewest duplicates.
489            pull_window: 4,
490        }
491    }
492}

623/// A message announced to us that we did not receive.

625struct Announced<PI> {
626    /// The time of the first announcement.
627    since: Instant,

630    /// Whether another peer announced it too.
631    by_others: bool,

634/// Returns how many of the `announced` messages were announced at least [`PULL_STALE_AFTER`] ago.
635///
636/// The messages that only one peer announced count at most [`MAX_STALE_PER_PEER`] for that peer,
637/// unless the peer is in `late_announcers`.
638fn stale<PI: PeerIdentity>(
639    announced: &HashMap<MessageId, Announced<PI>>,
640    late_announcers: &HashSet<PI>,
641    now: Instant,
642) -> usize {
643    let mut count = 0;
644    let mut alone: HashMap<PI, usize> = HashMap::new();
645    let stale = announced
646        .values()
647        .filter(|entry| now.saturating_duration_since(entry.since) >= PULL_STALE_AFTER);
648    for entry in stale {
649        if entry.by_others || late_announcers.contains(&entry.by) {
650            count += 1;
651        } else {
652            *alone.entry(entry.by).or_default() += 1;
653        }
654    }
655    count
656        + alone
657            .values()
658            .map(|n| (*n).min(MAX_STALE_PER_PEER))
659            .sum::<usize>()
660}

784    /// Handle an [`InEvent`].
785    pub fn handle(&mut self, event: InEvent<PI>, now: Instant, io: &mut impl IO<PI>) {

808    /// Panics if the per-peer bookkeeping disagrees with the trees.
809    ///
810    /// Runs after every event in the unit tests. Announcers in
811    /// `missing_messages` may be peers we do not hold yet: views converge in
812    /// their own time, and a graft makes them eager.
813    #[cfg(test)]
814    fn check_invariants(&self) {
815        assert!(self.trees.len() <= MAX_TREES);
816        for tree in self.trees.values() {
817            assert!(
818                tree.eager.is_disjoint(&tree.lazy),
819                "a peer is eager and lazy"
820            );
821            let all: BTreeSet<_> = tree.eager.union(&tree.lazy).copied().collect();
822            assert_eq!(all, self.neighbors, "a tree does not hold the neighbors");
823        }
824        for peer in self.lazy_push_queue.keys().chain(self.dropped.keys()) {
825            assert!(
826                self.neighbors.contains(peer),
827                "announcements queued for {peer:?}, which is no peer"
828            );
829        }
830        for (id, missing) in self.missing_messages.iter() {
831            assert!(
832                !missing.announcers.is_empty(),
833                "no announcer left for missing {id:?}"
834            );
835        }
836        for peer in self.pulling.keys() {
837            assert!(
838                self.neighbors.contains(peer),
839                "{peer:?} pulls, but is no peer"
840            );
841            assert!(
842                self.trees.values().all(|tree| !tree.eager.contains(peer)),
843                "{peer:?} pulls, but is eager"
844            );
845        }
846        if let Some(pull) = &self.pull {
847            assert!(pull.in_flight.len() <= self.config.pull_window);
848            assert!(pull.told.keys().all(|peer| self.neighbors.contains(peer)));
849            for (id, request) in pull.in_flight.iter() {
850                assert!(self.neighbors.contains(&request.peer));
851                // The peer we asked is no announcer until it announces again.
852                let asked_again = self.missing_messages.get(id).is_some_and(|missing| {
853                    missing
854                        .announcers
855                        .iter()
856                        .any(|(peer, _)| *peer == request.peer)
857                });
858                assert!(
859                    !asked_again,
860                    "{id:?} can be requested twice from {:?}",
861                    request.peer
862                );
863            }
864            for id in self.missing_messages.keys() {
865                assert!(
866                    pull.queue.contains(id) || pull.in_flight.contains_key(id),
867                    "missing {id:?} is never requested"
868                );
869            }
870        }

878    /// Handle receiving a [`Message`].
879    fn handle_message(&mut self, sender: PI, message: Message, now: Instant, io: &mut impl IO<PI>) {
880        if matches!(message, Message::Gossip(_)) {
881            self.stats.payload_messages_received += 1;
882        } else {
883            self.stats.control_messages_received += 1;
884        }
885        // A message from a non-neighbor is a straggler from a former one. Deliver
886        // its payload, but let it change no peer set.
887        let from_peer = self.neighbors.contains(&sender);
888        if from_peer {
889            self.tell_pull_mode(sender, &message, now, io);
890        }
891        match message {
892            Message::Gossip(details) => self.on_gossip(sender, from_peer, details, now, io),
893            _ if !from_peer => {}
894            Message::Prune(tree) => self.add_lazy(tree, sender, now),
895            Message::IHave(details) => {
896                self.on_ihave(sender, details, now, io);
897                if self.pull_check_at.is_none_or(|at| now >= at) {
898                    self.pull_check_at = Some(now + PULL_CHECK_INTERVAL);
899                    self.check_pull_enter(now, io);
900                }
901            }
902            Message::Graft(details) => self.on_graft(sender, details, now, io),
903            Message::Pull(mode) => self.on_pull(sender, mode),
904        }
905        self.pull_fill(now, io);
906    }

908    /// Dispatches messages from lazy queue over to lazy peers.
909    fn on_dispatch_timer(&mut self, io: &mut impl IO<PI>) {
910        let chunk_size = self.max_message_size
911            // Space for discriminator
912            - 1
913            // Space for the length of the list: 5 bytes hold any length that fits a frame
914            - 5;
915        let chunk_len = chunk_size / IHave::POSTCARD_MAX_SIZE;
916        while let Some((peer, list)) = self.lazy_push_queue.pop_first() {
917            if list.is_empty() {
918                continue;
919            }
920            // A batch that fits one frame can hand its allocation to the outgoing message.
921            if list.len() <= chunk_len {
922                self.send_ihave(peer, list, io);
923                continue;
924            }
925            for chunk in list.chunks(chunk_len) {
926                self.send_ihave(peer, chunk.to_vec(), io);
927            }
928        }
929
930        self.dispatch_timer_scheduled = false;
931    }

933    /// Send a gossip message.
934    ///
935    /// Will be pushed in full to eager peers.
936    /// Pushing the message id to the lazy peers is delayed by a timer.
937    fn broadcast(&mut self, content: Bytes, scope: Scope, now: Instant, io: &mut impl IO<PI>) {
938        // A message over the limit, at any hop, cannot go out, so we neither cache
939        // nor announce it.
940        if max_gossip_size(content.len(), scope) > self.max_message_size {
941            warn!(max = self.max_message_size, "broadcast too large, dropped");
942            return;
943        }
944        let scope = match scope {
945            Scope::Neighbors => DeliveryScope::Neighbors,
946            Scope::Swarm => DeliveryScope::Swarm(Round(0)),
947        };
948        let tree_id = self.tree_for_broadcast(scope, now);
949        let message = Gossip::new(content, scope, tree_id);
950        let id = message.id;
951        let me = self.me;
952        if let DeliveryScope::Swarm(_) = scope {
953            self.received_messages
954                .insert(id, now, now + self.config.message_id_retention);
955            self.cache.insert(
956                id,
957                message.clone(),
958                now + self.config.message_cache_retention,
959            );
960            self.lazy_push(&message, &me, now, io);
961        }
962
963        self.eager_push(&message, &me, &[], now, io);
964    }

966    /// Handle receiving a [`Message::Gossip`].
967    fn on_gossip(
968        &mut self,
969        sender: PI,
970        from_peer: bool,
971        message: Gossip,
972        now: Instant,
973        io: &mut impl IO<PI>,
974    ) {
975        // if we already received this message: move peer to lazy set
976        // and notify peer about this.
977        if self.received_messages.contains_key(&message.id) {
978            debug!(hash = %message.id.fmt_short(), "Received duplicate message... ignoring");
979            if message.round().is_none() || self.on_late_graft_reply(sender, &message, now, io) {
980                return;
981            }
982            // A non-neighbor that pushes to us may still hold us as an eager
983            // peer, as an older version does. The `Prune` stops its pushes.
984            if !from_peer || self.count_duplicate(sender, &message, now) {
985                self.add_lazy(message.tree, sender, now);
986                io.send(sender, Message::Prune(message.tree));
987            }
988        // otherwise store the message, emit to application and forward to peers
989        } else {
990            let verdict = self.verdict(sender, &message, now, io);
991            if let DeliveryScope::Swarm(prev_round) = message.scope {
992                // insert the message in the list of received messages
993                self.received_messages.insert(
994                    message.id,
995                    now,
996                    now + self.config.message_id_retention,
997                );
998                if let Some(entry) = self.announced.remove(&message.id) {
999                    let late = now.saturating_duration_since(entry.since) >= PULL_STALE_AFTER;
1000                    if late && entry.by != sender {
1001                        self.late_announcers.insert(entry.by);
1002                    }
1003                }
1004                if let Some(pull) = &mut self.pull {
1005                    pull.in_flight.remove(&message.id);
1006                }
1007                if verdict == Verdict::Reject {
1008                    // It counts as seen, and nothing more. We do not cache it, forward it,
1009                    // or graft for it.
1010                    self.graft_timer_scheduled.remove(&message.id);
1011                    self.missing_messages.remove(&message.id);
1012                    return;
1013                }
1014                // increase the round for forwarding the message, and add to cache
1015                // to reply to Graft messages later
1016                // TODO: add callback/event to application to get missing messages that were received before?
1017                let message = message.next_round().expect("just checked");
1018
1019                self.cache.insert(
1020                    message.id,
1021                    message.clone(),
1022                    now + self.config.message_cache_retention,
1023                );
1024                self.tree(message.tree, now, Some(message.id));
1025                // Push the message to our peers. A peer we grafted for it
1026                // has it already, and sends it to us.
1027                // A graft reply over no more hops can change the tree, see
1028                // `on_late_graft_reply`. The tree optimization would only see
1029                // the announcements left after the graft.
1030                let (grafted, graft_out) = match self.grafted.get_mut(&message.id) {
1031                    Some(grafts) => {
1032                        grafts.first = Some((sender, prev_round));
1033                        let out = grafts.peers.iter().any(|(_, round)| *round <= prev_round)
1034                            && !grafts.contains(&sender);
1035                        let peers = grafts.peers.iter().map(|(peer, _)| *peer).collect();
1036                        (peers, out)
1037                    }
1038                    None => (Vec::new(), false),
1039                };
1040                // A graft cannot make its peer eager in a tree that we do not know yet.
1041                for peer in &grafted {
1042                    self.add_eager(message.tree, *peer, now);
1043                }
1044                self.eager_push(&message, &sender, &grafted, now, io);
1045                self.lazy_push(&message, &sender, now, io);
1046                // cleanup places where we track missing messages
1047                self.graft_timer_scheduled.remove(&message.id);
1048                let previous_ihaves = self.missing_messages.remove(&message.id);
1049                // do the optimization step from the paper. In pull mode every copy comes
1050                // after an announcement, so the hops tell nothing about the tree.
1051                let optimize = from_peer && !graft_out && self.pull.is_none();
1052                if let Some(missing) = previous_ihaves.filter(|_| optimize) {
1053                    self.optimize_tree(&sender, &message, missing.announcers, now, io);
1054                }
1055                self.stats.max_last_delivery_hop = self
1056                    .stats
1057                    .max_last_delivery_hop
1058                    .max(prev_round.0.saturating_add(1));
1059            }
1060
1061            // emit event to application
1062            if verdict == Verdict::Accept {
1063                io.emit(Event::Received(GossipEvent::from_message(&message, sender)));
1064            }
1065        }
1066    }

1068    /// Counts a duplicate from the neighbor `sender`, and returns whether to prune it.
1069    ///
1070    /// The first duplicate from an eager neighbor prunes it, as in the paper, unless the
1071    /// message was first seen less than a graft timeout after our last prune in its tree. That
1072    /// message was in flight during the prune. Its duplicate may come from the cycle we just
1073    /// cut, and a second cut would split the tree. A link that alternates
1074    /// between first copy and duplicate is cut at its first duplicate. The duplicates of the
1075    /// message that created the tree come from its flood and always prune.
1076    fn count_duplicate(&mut self, sender: PI, message: &Gossip, now: Instant) -> bool {
1077        let first_seen = self.received_messages.get(&message.id).copied();
1078        let hold = self.config.graft_timeout_1;
1079        // A copy can name any tree, as the id covers only the content. Thus a copy creates
1080        // no tree.
1081        let Some(tree) = self.trees.get_mut(&message.tree) else {
1082            return true;
1083        };
1084        tree.last_used = now;
1085        if !tree.eager.contains(&sender) {
1086            return true;
1087        }
1088        let in_flight = tree.first_message != Some(message.id)
1089            && matches!((tree.last_prune, first_seen), (Some(prune), Some(seen)) if seen < prune + hold);
1090        if !in_flight {
1091            tree.last_prune = Some(now);
1092        }
1093        !in_flight
1094    }

1096    /// Returns the tree to name for a broadcast of ours, see [`Config::source_tree_after`].
1097    fn tree_for_broadcast(&mut self, scope: DeliveryScope, now: Instant) -> TreeId {
1098        if scope == DeliveryScope::Neighbors {
1099            return TreeId::default();
1100        }
1101        if self
1102            .last_sent
1103            .is_some_and(|last| last + TREE_RETENTION <= now)
1104        {
1105            self.sent = 0;
1106        }
1107        self.last_sent = Some(now);
1108        let own = self.sent >= self.config.source_tree_after;
1109        self.sent = self.sent.saturating_add(1);
1110        if own {
1111            TreeId::of(&self.me)
1112        } else {
1113            TreeId::default()
1114        }
1115    }

1117    /// Returns the verdict on a new message, and asks the filter at most once per message id.
1118    ///
1119    /// The state does not deduplicate messages to the neighbors, so it keeps the first verdict
1120    /// for their copies. A later copy to the swarm also gets that verdict.
1121    fn verdict(
1122        &mut self,
1123        sender: PI,
1124        message: &Gossip,
1125        now: Instant,
1126        io: &mut impl IO<PI>,
1127    ) -> Verdict {
1128        if let Some(verdict) = self.neighbor_verdicts.get(&message.id) {
1129            return *verdict;
1130        }
1131        let Some(verdict) = io.filter(message, sender) else {
1132            return Verdict::Accept;
1133        };
1134        if message.scope == DeliveryScope::Neighbors {
1135            let expires = now + self.config.message_id_retention;
1136            self.neighbor_verdicts.insert(message.id, verdict, expires);
1137        }
1138        verdict
1139    }

1141    /// Handles a duplicate that a peer we grafted sends after the eager copy came.
1142    ///
1143    /// The peer announced the message more than a graft timeout before the
1144    /// eager copy arrived, so its path is faster. If its copy took no more
1145    /// hops, it stays eager, and the sender of the eager copy turns lazy.
1146    /// Otherwise the graft and its duplicate would repeat for each message.
1147    /// The peer is not below us, as it announced the message before we had it.
1148    /// The hop check keeps [`Self::optimize_tree`] from undoing the change on
1149    /// the next message. A reply to a retried graft is ignored.
1150    ///
1151    /// Returns whether it handled the duplicate.
1152    fn on_late_graft_reply(
1153        &mut self,
1154        sender: PI,
1155        message: &Gossip,
1156        now: Instant,
1157        io: &mut impl IO<PI>,
1158    ) -> bool {
1159        let Some(grafts) = self.grafted.get_mut(&message.id) else {
1160            return false;
1161        };
1162        let Some((first, first_round)) = grafts.first else {
1163            return false;
1164        };
1165        if !grafts.contains(&sender) {
1166            return false;
1167        }
1168        // A retry brings a second reply from a peer that already gave us the message.
1169        if first == sender || grafts.kept == Some(sender) {
1170            return true;
1171        }
1172        let no_more_hops = message.round().is_some_and(|round| round <= first_round);
1173        let both_eager = self
1174            .trees
1175            .get(&message.tree)
1176            .is_some_and(|tree| tree.eager.contains(&sender) && tree.eager.contains(&first));
1177        if !no_more_hops || grafts.contains(&first) || !both_eager {
1178            return false;
1179        }
1180        grafts.kept = Some(sender);
1181        debug!(peer = ?sender, slower = ?first, "keep a faster grafted peer");
1182        self.add_lazy(message.tree, first, now);
1183        io.send(first, Message::Prune(message.tree));
1184        true
1185    }

1187    /// Optimize the tree by pruning the `sender` of a [`Message::Gossip`] if we previously
1188    /// received a [`Message::IHave`] for the same message with a much lower number of delivery
1189    /// hops from the original broadcaster of the message.
1190    ///
1191    /// See [Config::optimization_threshold].
1192    fn optimize_tree(
1193        &mut self,
1194        gossip_sender: &PI,
1195        message: &Gossip,
1196        previous_ihaves: VecDeque<(PI, Round)>,
1197        now: Instant,
1198        io: &mut impl IO<PI>,
1199    ) {
1200        let round = message.round().expect("only called for swarm messages");
1201        let best_ihave = previous_ihaves
1202            .iter()
1203            .min_by(|(_a_peer, a_round), (_b_peer, b_round)| a_round.cmp(b_round))
1204            .copied();
1205
1206        if let Some((ihave_peer, ihave_round)) = best_ihave {
1207            if (ihave_round < round) && (round - ihave_round) >= self.config.optimization_threshold
1208            {
1209                // Graft the sender of the IHave, but only if it's not already eager.
1210                let tree = message.tree;
1211                if !self
1212                    .tree(tree, now, Some(message.id))
1213                    .eager
1214                    .contains(&ihave_peer)
1215                {
1216                    let message = Message::Graft(Graft { id: None, tree });
1217                    self.add_eager(tree, ihave_peer, now);
1218                    io.send(ihave_peer, message);
1219                }
1220                // Prune the sender of the Gossip.
1221                self.add_lazy(tree, *gossip_sender, now);
1222                io.send(*gossip_sender, Message::Prune(tree));
1223            }
1224        }
1225    }

1227    /// Handle receiving a [`Message::IHave`].
1228    ///
1229    /// > When a node receives a IHAVE message, it simply marks the corresponding message as
1230    /// > missing It then starts a timer, with a predefined timeout value, and waits for the missing
1231    /// > message to be received via eager push before the timer expires. The timeout value is a
1232    /// > protocol parameter that should be configured considering the diameter of the overlay and a
1233    /// > target maximum recovery latency, defined by the application requirements. This is a
1234    /// > parameter that should be statically configured at deployment time. (p8)
1235    fn on_ihave(&mut self, sender: PI, ihaves: Vec<IHave>, now: Instant, io: &mut impl IO<PI>) {
1236        for ihave in ihaves {
1237            if !self.received_messages.contains_key(&ihave.id) {
1238                let announced = self.announced.entry(ihave.id).or_insert(Announced {
1239                    since: now,
1240                    by: sender,
1241                    by_others: false,
1242                });
1243                announced.by_others |= announced.by != sender;
1244                let is_new = !self.missing_messages.contains_key(&ihave.id);
1245                // Queue each peer once, as it answers every graft. A new
1246                // announcement after a graft is a retry, for example after a
1247                // dropped reply.
1248                let missing = self
1249                    .missing_messages
1250                    .entry(ihave.id)
1251                    .or_insert_with(|| Missing {
1252                        announcers: VecDeque::new(),
1253                        tree: ihave.tree,
1254                    });
1255                if !missing.announcers.iter().any(|(peer, _)| *peer == sender) {
1256                    missing.announcers.push_back((sender, ihave.round));
1257                }
1258
1259                if let Some(pull) = &mut self.pull {
1260                    // The window decides when to ask, not a timer.
1261                    let asked = pull.in_flight.get(&ihave.id).map(|request| request.peer);
1262                    if asked == Some(sender) {
1263                        // The peer announces again what its queue dropped, so our
1264                        // request gets no reply.
1265                        pull.in_flight.remove(&ihave.id);
1266                        pull.queue.push_front(ihave.id);
1267                    } else if is_new {
1268                        pull.queue.push_back(ihave.id);
1269                    }
1270                } else if !self.graft_timer_scheduled.contains(&ihave.id) {
1271                    self.graft_timer_scheduled.insert(ihave.id);
1272                    io.schedule(self.config.graft_timeout_1, Timer::SendGraft(ihave.id));
1273                }
1274            }
1275        }
1276    }

1278    /// A scheduled [`Timer::SendGraft`] has reached it's deadline.
1279    fn on_send_graft_timer(&mut self, id: MessageId, now: Instant, io: &mut impl IO<PI>) {
1280        self.graft_timer_scheduled.remove(&id);
1281        if self.pull.is_some() {
1282            return;
1283        }
1284        // if the message was received before the timer ran out, there is no need to request it
1285        // again
1286        if self.received_messages.contains_key(&id) {
1287            return;
1288        }
1289        // get the first peer that advertised this message
1290        let entry = self
1291            .missing_messages
1292            .get_mut(&id)
1293            .and_then(|missing| Some((missing.tree, missing.announcers.pop_front()?)));
1294        if self
1295            .missing_messages
1296            .get(&id)
1297            .is_some_and(|missing| missing.announcers.is_empty())
1298        {
1299            self.missing_messages.remove(&id);
1300        }
1301        if let Some((tree, (peer, round))) = entry {
1302            self.add_eager(tree, peer, now);
1303            match self.grafted.get_mut(&id) {
1304                Some(grafts) => match grafts.peers.iter_mut().find(|(p, _)| *p == peer) {
1305                    Some((_, known)) => *known = (*known).min(round),
1306                    None => grafts.peers.push((peer, round)),
1307                },
1308                None => {
1309                    let grafts = Grafts {
1310                        peers: vec![(peer, round)],
1311                        first: None,
1312                        kept: None,
1313                    };
1314                    let expires = now + self.config.message_cache_retention;
1315                    self.grafted.insert(id, grafts, expires);
1316                }
1317            }
1318            let message = Message::Graft(Graft { id: Some(id), tree });
1319            io.send(peer, message);
1320
1321            // "when a GRAFT message is sent, another timer is started to expire after a certain timeout,
1322            // to ensure that the message will be requested to another neighbor if it is not received
1323            // meanwhile. This second timeout value should be smaller that the first, in the order of
1324            // an average round trip time to a neighbor." (p9)
1325            // A new announcement does not start a second chain of grafts meanwhile.
1326            self.graft_timer_scheduled.insert(id);
1327            io.schedule(self.config.graft_timeout_2, Timer::SendGraft(id));
1328        }
1329    }

1331    /// Enters pull mode if at least [`PULL_ENTER_STALE`] messages are stale.
1332    ///
1333    /// The first [`PULL_SOURCES`] neighbors keep announcing to us, the others go silent.
1334    fn check_pull_enter(&mut self, now: Instant, io: &mut impl IO<PI>) {
1335        if self.pull.is_some() || self.config.pull_window == 0 {
1336            return;
1337        }
1338        let stale = stale(&self.announced, &self.late_announcers, now);
1339        if stale < PULL_ENTER_STALE {
1340            return;
1341        }
1342        info!(me = ?self.me, stale, "enter pull mode");
1343        let mut ids: Vec<_> = self
1344            .missing_messages
1345            .keys()
1346            .map(|id| (self.announced.get(id).map(|entry| entry.since), *id))
1347            .collect();
1348        ids.sort_by_key(|(since, _)| *since);
1349        self.pull = Some(Pull {
1350            queue: ids.into_iter().map(|(_, id)| id).collect(),
1351            in_flight: HashMap::new(),
1352            told: BTreeMap::new(),
1353            caught_up: None,
1354        });
1355        self.unsent_pull_off.clear();
1356        for (i, peer) in self.neighbors.clone().into_iter().enumerate() {
1357            let mode = match i < PULL_SOURCES {
1358                true => PullMode::Source,
1359                false => PullMode::Silent,
1360            };
1361            self.tell_pull(peer, mode, now, io);
1362        }
1363        io.schedule(PULL_TICK, Timer::PullTick);
1364        self.pull_fill(now, io);
1365    }

1385    /// Tells the neighbor `peer` our pull mode if it did not learn it.
1386    ///
1387    /// A peer learns it on its first message after we entered. A `Pull` that we send
1388    /// when we make it a neighbor can arrive before it makes us one, and it then ignores
1389    /// the `Pull`. A peer that pushes a message we did not request, or announces while
1390    /// silent, may have lost the `Pull` to a full queue. It hears it again after
1391    /// [`PULL_REQUEST_TIMEOUT`], as until then it may send what it queued before.
1392    fn tell_pull_mode(&mut self, peer: PI, message: &Message, now: Instant, io: &mut impl IO<PI>) {
1393        let Some(pull) = &self.pull else {
1394            return;
1395        };
1396        let Some(&(mode, at)) = pull.told.get(&peer) else {
1397            self.tell_pull(peer, PullMode::Silent, now, io);
1398            return;
1399        };
1400        let unwanted = match message {
1401            Message::Gossip(gossip) => {
1402                gossip.round().is_some()
1403                    && pull
1404                        .in_flight
1405                        .get(&gossip.id)
1406                        .is_none_or(|request| request.peer != peer)
1407            }
1408            Message::IHave(_) => mode == PullMode::Silent,
1409            _ => false,
1410        };
1411        if unwanted && now >= at + PULL_REQUEST_TIMEOUT {
1412            self.tell_pull(peer, mode, now, io);
1413        }
1414    }

1416    /// Sends requests until the window is full.
1417    ///
1418    /// A request is a `Graft` for one message, with the tree of the announcement, to the
1419    /// first of its announcers with the fewest requests open. Like a `Graft` on the timer,
1420    /// it removes the announcer, so no peer gets a second request for a message unless it
1421    /// announces it again. The peer does not make us eager, see [`Self::on_pull`].
1422    fn pull_fill(&mut self, now: Instant, io: &mut impl IO<PI>) {
1423        let Some(pull) = &mut self.pull else {
1424            return;
1425        };
1426        while pull.in_flight.len() < self.config.pull_window {
1427            let Some(id) = pull.queue.pop_front() else {
1428                break;
1429            };
1430            if pull.in_flight.contains_key(&id) {
1431                continue;
1432            }
1433            let Some(missing) = self.missing_messages.get_mut(&id) else {
1434                continue;
1435            };
1436            let tree = missing.tree;
1437            // A peer with a long queue to us would otherwise hold the whole window.
1438            let open = |peer: &PI| {
1439                let requests = pull.in_flight.values();
1440                requests.filter(|request| request.peer == *peer).count()
1441            };
1442            let (peer, round) = missing
1443                .announcers
1444                .iter()
1445                .enumerate()
1446                .min_by_key(|(_, (peer, _))| open(peer))
1447                .map(|(i, _)| i)
1448                .and_then(|i| missing.announcers.remove(i))
1449                .expect("a missing message has an announcer");
1450            if missing.announcers.is_empty() {
1451                self.missing_messages.remove(&id);
1452            }
1453            let graft = Message::Graft(Graft { id: Some(id), tree });
1454            if io.send(peer, graft).is_some() {
1455                // The queue to the peer is full. The next event tries again.
1456                self.missing_messages
1457                    .entry(id)
1458                    .or_insert_with(|| Missing {
1459                        announcers: VecDeque::new(),
1460                        tree,
1461                    })
1462                    .announcers
1463                    .push_front((peer, round));
1464                pull.queue.push_front(id);
1465                break;
1466            }
1467            pull.in_flight.insert(id, Request { peer, sent: now });
1468        }
1469    }

1471    /// Expires requests, replaces sources that left, and leaves pull mode once we keep up.
1472    fn on_pull_tick(&mut self, now: Instant, io: &mut impl IO<PI>) {
1473        let window = self.config.pull_window;
1474        let Some(pull) = &mut self.pull else {
1475            return;
1476        };
1477        let mut expired: Vec<_> = pull
1478            .in_flight
1479            .iter()
1480            .filter(|(_, request)| now >= request.sent + PULL_REQUEST_TIMEOUT)
1481            .map(|(id, request)| (request.sent, *id))
1482            .collect();
1483        // The oldest goes to the front last, so it is the first again.
1484        expired.sort_by_key(|(sent, _)| *sent);
1485        for (_, id) in expired.into_iter().rev() {
1486            pull.in_flight.remove(&id);
1487            // The reply may still come, so the peer we asked gets no second request.
1488            if self.missing_messages.contains_key(&id) {
1489                pull.queue.push_front(id);
1490            }
1491        }
1492        // A message with no announcer left to ask and no request open is no backlog we can
1493        // work off. Its copy may still come, but it would keep us in pull mode for the
1494        // whole cache retention.
1495        let missing = &self.missing_messages;
1496        self.announced.retain(|id, entry| {
1497            now < entry.since + PULL_REQUEST_TIMEOUT
1498                || missing.contains_key(id)
1499                || pull.in_flight.contains_key(id)
1500        });
1501        if stale(&self.announced, &self.late_announcers, now) > window {
1502            pull.caught_up = None;
1503        } else if now >= *pull.caught_up.get_or_insert(now) + PULL_EXIT_AFTER {
1504            self.leave_pull(io);
1505            return;
1506        }
1507        let sources = pull
1508            .told
1509            .values()
1510            .filter(|(mode, _)| *mode == PullMode::Source)
1511            .count();
1512        let more: Vec<_> = self
1513            .neighbors
1514            .iter()
1515            .filter(|peer| {
1516                pull.told
1517                    .get(*peer)
1518                    .is_none_or(|(mode, _)| *mode != PullMode::Source)
1519            })
1520            .take(PULL_SOURCES.saturating_sub(sources))
1521            .copied()
1522            .collect();
1523        for peer in more {
1524            self.tell_pull(peer, PullMode::Source, now, io);
1525        }
1526        io.schedule(PULL_TICK, Timer::PullTick);
1527        self.pull_fill(now, io);
1528    }

1549    ///

1594    /// Forgets a neighbor that left the topic.
1595    ///
1596    /// > When a neighbor is detected to leave the overlay, it is simple removed from the
1597    /// > membership. Furthermore, the record of IHAVE messages sent from failed members is deleted
1598    /// > from the missing history. (p9)
1599    pub(super) fn on_neighbor_down(&mut self, peer: PI) {
1600        self.missing_messages.retain(|_message_id, missing| {
1601            missing
1602                .announcers
1603                .retain(|(ihave_peer, _round)| *ihave_peer != peer);
1604            !missing.announcers.is_empty()
1605        });
1606        self.neighbors.remove(&peer);
1607        for tree in self.trees.values_mut() {
1608            tree.eager.remove(&peer);
1609            tree.lazy.remove(&peer);
1610        }
1611        // The dispatch timer would otherwise send these, and the network layer
1612        // would dial the peer to deliver them.
1613        self.lazy_push_queue.remove(&peer);
1614        self.dropped.remove(&peer);
1615        // A peer that comes back has lost its state, so it needs our pushes again.
1616        for grafts in self.grafted.values_mut() {
1617            grafts.peers.retain(|(grafted, _)| *grafted != peer);
1618        }
1619        self.pulling.remove(&peer);
1620        self.unsent_pull_off.remove(&peer);
1621        self.late_announcers.remove(&peer);
1622        if let Some(pull) = &mut self.pull {
1623            pull.told.remove(&peer);
1624            let missing = &self.missing_messages;
1625            pull.in_flight.retain(|id, request| {
1626                if request.peer != peer {
1627                    return true;
1628                }
1629                if missing.contains_key(id) {
1630                    pull.queue.push_front(*id);
1631                }
1632                false
1633            });
1634        }
1635        // A message that no neighbor can send us any more is no backlog.
1636        let pull = self.pull.as_ref();
1637        self.announced.retain(|id, _| {
1638            self.missing_messages.contains_key(id)
1639                || pull.is_some_and(|pull| pull.in_flight.contains_key(id))
1640                || self
1641                    .grafted
1642                    .get(id)
1643                    .is_some_and(|grafts| !grafts.peers.is_empty())
1644        });
1645    }

1656    /// Notes a dropped push or announcement, to announce it to the peer later.
1657    ///
1658    /// A dropped neighbor message is lost, as we do not cache it.
1659    fn note_dropped(
1660        &mut self,
1661        peer: PI,
1662        ihaves: impl IntoIterator<Item = IHave>,
1663        io: &mut impl IO<PI>,
1664    ) {
1665        if !self.neighbors.contains(&peer) || self.pulling.get(&peer) == Some(&PullMode::Silent) {
1666            return;
1667        }
1668        let mut total: usize = self.dropped.values().map(IndexMap::len).sum();
1669        let ids = self.dropped.entry(peer).or_default();
1670        for IHave { id, round, tree } in ihaves {
1671            if !self.cache.contains_key(&id) || ids.contains_key(&id) {
1672                continue;
1673            }
1674            // Past the caps, the peer misses the message unless another peer has it.
1675            if ids.len() >= MAX_DROPPED_PER_PEER || total >= MAX_DROPPED {
1676                debug!("too many dropped messages to announce again, drop {id}");
1677                continue;
1678            }
1679            ids.insert(id, (round, tree));
1680            total += 1;
1681        }
1682        if ids.is_empty() {
1683            self.dropped.remove(&peer);
1684        }
1685        self.schedule_announce_dropped(io);
1686    }

1711        if !self.lazy_push_queue.is_empty() {

1717    fn on_evict_cache_timer(&mut self, now: Instant, io: &mut impl IO<PI>) {
1718        self.cache.expire_until(now);
1719        self.received_messages.expire_until(now);
1720        self.neighbor_verdicts.expire_until(now);
1721        self.grafted.expire_until(now);
1722        if let Some(shared) = self.trees.get_mut(&TreeId::default()) {
1723            shared.last_used = now;
1724        }
1725        self.trees
1726            .retain(|_, tree| tree.last_used + TREE_RETENTION > now);
1727        // No peer caches an older message, so nobody can send it.
1728        let retention = self.config.message_cache_retention;
1729        let missing = &mut self.missing_messages;
1730        self.announced.retain(|id, entry| {
1731            let keep = entry.since + retention > now;
1732            if !keep {
1733                missing.remove(id);
1734            }
1735            keep
1736        });
1737        if let Some(pull) = &mut self.pull {
1738            pull.queue.retain(|id| missing.contains_key(id));
1739        }
1740        for peer in std::mem::take(&mut self.unsent_pull_off) {
1741            self.send_pull(peer, PullMode::Off, io);
1742        }
1743        self.check_pull_enter(now, io);
1744        io.schedule(self.config.cache_evict_interval, Timer::EvictCache);

1747    /// Returns our part of the tree `id`, and creates the tree if needed.
1748    ///
1749    /// A new tree is a copy of the shared tree, or has every neighbor eager if there is no
1750    /// shared tree.
1751    ///
1752    /// Past [`MAX_TREES`], the least recently used tree makes room.
1753    /// `first_message` is the message that creates the tree.
1754    fn tree(
1755        &mut self,
1756        id: TreeId,
1757        now: Instant,
1758        first_message: Option<MessageId>,
1759    ) -> &mut Tree<PI> {
1760        if !self.trees.contains_key(&id) && self.trees.len() >= MAX_TREES {
1761            let oldest = self
1762                .trees
1763                .iter()
1764                .filter(|(id, _)| **id != TreeId::default())
1765                .min_by_key(|(_, tree)| tree.last_used);
1766            if let Some(oldest) = oldest.map(|(id, _)| *id) {
1767                self.trees.remove(&oldest);
1768            }
1769        }
1770        if !self.trees.contains_key(&id) {
1771            // The copy makes the switch of a sender to its own tree cost no flood.
1772            let (mut eager, mut lazy) = match self.trees.get(&TreeId::default()) {
1773                Some(shared) => (shared.eager.clone(), shared.lazy.clone()),
1774                None => (self.neighbors.clone(), BTreeSet::new()),
1775            };
1776            for peer in self.pulling.keys() {
1777                eager.remove(peer);
1778                lazy.insert(*peer);
1779            }
1780            let tree = Tree {
1781                eager,
1782                lazy,
1783                last_used: now,
1784                last_prune: None,
1785                first_message,
1786            };
1787            self.trees.insert(id, tree);
1788        }
1789        let tree = self.trees.get_mut(&id).expect("exists or just inserted");
1790        tree.last_used = now;
1791        tree
1792    }

1794    /// Moves a neighbor into the eager set of a known tree.
1795    ///
1796    /// It ignores a peer that is not a neighbor. A `Graft`, `Prune`, or
1797    /// `IHave` can still be in flight when the membership layer drops a peer,
1798    /// and handling it must not put the peer back: we handled its `NeighborDown`
1799    /// already, so nothing would remove the peer again.
1800    ///
1801    /// Only a payload creates a tree, see [`Self::tree`]. A peer could otherwise
1802    /// make up tree ids in small control messages, and evict the real trees.
1803    ///
1804    /// A peer in pull mode stays lazy.
1805    fn add_eager(&mut self, tree: TreeId, peer: PI, now: Instant) {
1806        if self.pulling.contains_key(&peer) {
1807            return;
1808        }
1809        if let Some(tree) = self.known_tree(tree, &peer, now) {
1810            tree.lazy.remove(&peer);
1811            tree.eager.insert(peer);
1812        }
1813    }
1814
1815    /// Moves a neighbor into the lazy set of a known tree, see [`Self::add_eager`].
1816    fn add_lazy(&mut self, tree: TreeId, peer: PI, now: Instant) {
1817        if let Some(tree) = self.known_tree(tree, &peer, now) {
1818            tree.eager.remove(&peer);
1819            tree.lazy.insert(peer);
1820        }
1821    }
1822
1823    /// Returns the tree `id` if `peer` is a neighbor and the tree exists or is the shared one.
1824    fn known_tree(&mut self, id: TreeId, peer: &PI, now: Instant) -> Option<&mut Tree<PI>> {
1825        if !self.neighbors.contains(peer) {
1826            return None;
1827        }
1828        if id == TreeId::default() {
1829            return Some(self.tree(id, now, None));
1830        }
1831        let tree = self.trees.get_mut(&id)?;
1832        tree.last_used = now;
1833        Some(tree)
1834    }
1835
1836    /// Sends `gossip` at once to the eager peers of its tree.
1837    ///
1838    /// A message to the neighbors goes to all neighbors.
1839    /// A peer whose queue dropped the message gets it announced later.
1840    fn eager_push(
1841        &mut self,
1842        gossip: &Gossip,
1843        sender: &PI,
1844        skip: &[PI],
1845        now: Instant,
1846        io: &mut impl IO<PI>,
1847    ) {
1848        let me = self.me;
1849        let peers = match gossip.scope {
1850            DeliveryScope::Neighbors => &self.neighbors,
1851            DeliveryScope::Swarm(_) => &self.tree(gossip.tree, now, Some(gossip.id)).eager,
1852        };
1853        let mut dropped = Vec::new();
1854        for peer in peers
1855            .iter()
1856            .filter(|peer| **peer != me && *peer != sender && !skip.contains(peer))
1857        {
1858            if io.send(*peer, Message::Gossip(gossip.clone())).is_some() {
1859                dropped.push(*peer);
1860            }
1861        }
1862        if let Some(ihave) = gossip.ihave() {
1863            for peer in dropped {
1864                self.note_dropped(peer, [ihave.clone()], io);
1865            }
1866        }
1867    }
1868
1869    /// Queue lazy message announcements into the queue that will be sent out as batched
1870    /// [`Message::IHave`] messages once the [`Timer::DispatchLazyPush`] timer is triggered.
1871    fn lazy_push(&mut self, gossip: &Gossip, sender: &PI, now: Instant, io: &mut impl IO<PI>) {
1872        let Some(ihave) = gossip.ihave() else {
1873            return;
1874        };
1875        self.tree(gossip.tree, now, Some(gossip.id));
1876        let pulling = &self.pulling;
1877        for peer in self.trees[&gossip.tree]
1878            .lazy
1879            .iter()
1880            .filter(|x| *x != sender && pulling.get(*x) != Some(&PullMode::Silent))
1881        {
1882            self.lazy_push_queue
1883                .entry(*peer)
1884                .or_default()
1885                .push(ihave.clone());
1886        }
1887        self.schedule_dispatch(io);
1888    }
1889
1890    fn schedule_dispatch(&mut self, io: &mut impl IO<PI>) {
1891        if !self.dispatch_timer_scheduled {
1892            io.schedule(self.config.dispatch_timeout, Timer::DispatchLazyPush);
1893            self.dispatch_timer_scheduled = true;
1894        }
1895    }
