1//! The send and receive loops of a connection, around its control stream.
2//!
3//! The dialer opens the control stream and writes its header. The acceptor
4//! answers with its own. After that the stream carries only `GOAWAY`, with the
5//! number of topic streams its sender opened. Either writer sends it when the
6//! connection is idle, or when it got the `GOAWAY` of the peer. A side finishes
7//! its half of the stream once it read all streams that the peer counted, and
8//! closes the connection once the peer finished its half too.

91/// A slot for one value, which a later value replaces.
92///
93/// No code keeps the lock across an await.
94#[derive(Debug)]
95pub(super) struct Mailbox<T> {
96    value: Mutex<Option<T>>,
97    notify: Notify,
98}
99
100impl<T> Default for Mailbox<T> {
101    fn default() -> Self {
102        Self {
103            value: Mutex::new(None),
104            notify: Notify::new(),
105        }
106    }
107}
108
109impl<T> Mailbox<T> {
110    /// Puts `value` in the slot, and drops the value that nobody took.
111    pub(super) fn put(&self, value: T) {
112        let old = self.value.lock().expect("poisoned").replace(value);
113        drop(old);
114        self.notify.notify_one();
115    }
116
117    /// Takes the value, if the slot has one.
118    pub(super) fn try_take(&self) -> Option<T> {
119        self.value.lock().expect("poisoned").take()
120    }
121
122    /// Waits for a value and takes it.
123    ///
124    /// It is cancellation-safe: a value that arrives stays in the slot.
125    async fn take(&self) -> T {
126        loop {
127            if let Some(value) = self.try_take() {
128                return value;
129            }
130            // A put after the check above leaves a permit, so this does not miss it.
131            self.notify.notified().await;
132        }
133    }
134}

178/// Returns a finished task's output, or `None` if the runtime cancelled the task.
179///
180/// A task cancelled at runtime shutdown is not a bug, so it must not panic the actor (#140).
181/// A panic in the task panics the caller with the same payload.
182pub(super) fn task_output<T>(res: Result<T, n0_future::task::JoinError>) -> Option<T> {
183    match res {
184        Ok(output) => Some(output),
185        Err(err) if err.is_cancelled() => None,
186        #[cfg(not(target_family = "wasm"))]
187        Err(err) => std::panic::resume_unwind(err.into_panic()),
188        #[cfg(target_family = "wasm")]
189        Err(err) => panic!("task failed: {err}"),
190    }
191}

193/// The reason why a connection fails, and the code that it closes with.
194#[derive(Debug)]
195struct Failure {
196    code: CloseCode,
197    reason: String,
198}
199
200impl Failure {
201    fn new(code: CloseCode, reason: impl Into<String>) -> Self {
202        Self {
203            code,
204            reason: reason.into(),
205        }
206    }
207
208    fn protocol(reason: impl Into<String>) -> Self {
209        Self::new(CloseCode::Protocol, reason)
210    }
211
212    /// The actor stopped. The peer must not read a quiet close, which keeps us as a neighbor.
213    fn actor_gone() -> Self {
214        Self::new(CloseCode::Shutdown, "dropped by the actor")
215    }
216}

218impl From<WriteError> for Failure {
219    fn from(err: WriteError) -> Self {
220        let code = match err {
221            WriteError::Stalled { .. } | WriteError::OpenStalled { .. } => CloseCode::Blocked,
222            _ => CloseCode::SendFailed,
223        };
224        Self::new(code, err.to_string())
225    }
226}

258/// Runs the I/O of a connection until it ends, and returns how.
259///
260/// The connection ends when a loop fails, when the peer state aborts it, or
261/// after a finished retire. The task closes it with the code of the failure.
262/// It also returns the reader of the queue, if the writer still had it.
263pub(super) async fn connection_loop(args: ConnArgs) -> ConnExit {
264    let conn = args.conn.clone();
265    debug!(side = ?conn.side(), "connection established");
266    let aborted = args.aborted.clone();
267    // Outside the loops, so that the reader stays when a loop fails.
268    let mut reader = None;
269    let result = tokio::select! {
270        biased;
271        _ = aborted.cancelled() => Err(Failure::new(CloseCode::Blocked, "send queue full")),
272        res = run_loops(args, &mut reader) => res,
273    };
274    let end = match result {
275        Ok(end) => end,
276        Err(failure) => match conn.close_reason() {
277            Some(reason) => end_of(Some(reason)),
278            None => {
279                debug!(code = ?failure.code, "close the connection: {}", failure.reason);
280                conn.close(failure.code.code(), failure.reason.as_bytes());
281                ConnEnd::Failed
282            }
283        },
284    };
285    ConnExit { end, reader }
286}

288/// Returns how a connection ended, from its close reason.
289fn end_of(reason: Option<ConnectionError>) -> ConnEnd {
290    match reason {
291        Some(ConnectionError::ApplicationClosed(close)) => {
292            match CloseCode::from_code(close.error_code) {
293                Some(CloseCode::Retired) => ConnEnd::Retired,
294                Some(CloseCode::Stale) => ConnEnd::Quiet,
295                _ => ConnEnd::Failed,
296            }
297        }
298        _ => ConnEnd::Failed,
299    }
300}

302/// Runs the control handshake and both loops, until both end.
303///
304/// The send loop keeps the reader of the queue in `reader` while it writes.
305async fn run_loops(args: ConnArgs, reader: &mut Option<QueueReader>) -> Result<ConnEnd, Failure> {
306    let ConnArgs {
307        id,
308        conn,
309        retire,
310        take_queue,
311        shared,
312        ..
313    } = args;
314    let ConnShared {
315        events,
316        max_message_size,
317        verifier,
318        session,
319        retire_after,
320        shutdown,
321    } = shared;
322    let dialer = conn.side().is_client();
323    let events = EventSink {
324        tx: events,
325        peer: conn.remote_id(),
326        id,
327    };
328    let (mut ctl_tx, ctl_rx) = if dialer {
329        // A peer that grants no bidirectional stream keeps the open pending.
330        time::timeout(HEADER_TIMEOUT, conn.open_bi())
331            .await
332            .map_err(|_| Failure::protocol("cannot open the control stream in time"))?
333            .map_err(|err| Failure::protocol(format!("cannot open the control stream: {err:#}")))?
334    } else {
335        time::timeout(HEADER_TIMEOUT, conn.accept_bi())
336            .await
337            .map_err(|_| Failure::protocol("no control stream"))?
338            .map_err(|err| Failure::protocol(format!("no control stream: {err:#}")))?
339    };
340    write_all(&mut ctl_tx, &session.to_frame()).await?;
341    let signals = Arc::new(Signals::default());
342    let recv_loop = RecvLoop {
343        conn: conn.clone(),
344        events: events.clone(),
345        max_message_size,
346        verifier,
347        rejected_topics: Default::default(),
348        signals: signals.clone(),
349    };
350    let send_loop = SendLoop {
351        conn: conn.clone(),
352        events,
353        signals,
354        ctl: ctl_tx,
355        retire,
356        take_queue,
357        shutdown,
358        retire_after,
359        max_message_size,
360        streams: Default::default(),
361        stream_ends: Default::default(),
362        buffer: Default::default(),
363        topics_to_open: Default::default(),
364        opened_count: 0,
365        refused_topics: Default::default(),
366        retiring_until: None,
367    };
368    let send = async { send_loop.run(reader).await.map_err(Failure::from) }
369        .instrument(error_span!("send"));
370    let recv = async {
371        let mut ctl_rx = ctl_rx;
372        recv_loop.read_header(&mut ctl_rx).await?;
373        recv_loop.run(ctl_rx).await
374    }
375    .instrument(error_span!("recv"));
376    tokio::try_join!(send, recv).map(|(_, end)| end)
377}

379/// Reads the streams that the peer opens on a connection.
380struct RecvLoop {
381    conn: Connection,
382    events: EventSink,
383    max_message_size: usize,
384    verifier: TopicAuth,
385    /// The topics whose token the verifier rejected on this connection, at most [`MAX_REJECTED_TOPICS`].
386    ///
387    /// The loop stops a later stream of such a topic, and does not call the verifier.
388    rejected_topics: HashSet<TopicId>,
389    signals: Arc<Signals>,
390}
391
392/// The result of a stream read: the stream, and its next message or its end.
393type ReadResult = (TopicReader, Result<Option<ProtoMessage>, ReadError>);
394
395impl RecvLoop {
396    /// Reads the control header of the peer, and tells the actor its session.
397    async fn read_header(&self, ctl: &mut RecvStream) -> Result<(), Failure> {
398        let session = Session::read(ctl, &mut Vec::new())
399            .await
400            .map_err(|err| Failure::protocol(format!("control header: {err:#}")))?;
401        self.events.send(ConnEventKind::Header(session)).await?;
402        Ok(())
403    }
404
405    /// Reads the streams of the connection until it closes, or until a retire finished.
406    ///
407    /// The loop reads one header at a time, so streams decode in the order of
408    /// their ids. A newer stream of a topic waits until the older one ended. So
409    /// the actor sees the messages of a topic in the order the peer wrote them.
410    /// A late header, a bad header, a bad frame, or a reset stream fails the
411    /// connection.
412    async fn run(mut self, mut ctl: RecvStream) -> Result<ConnEnd, Failure> {
413        // The header of the stream that we accepted last, while we read it.
414        let mut header: Option<BoxFuture<Result<TopicReader, ReadError>>> = None;
415        // The next message of each stream that we read.
416        let mut reads: FuturesUnordered<BoxFuture<ReadResult>> = FuturesUnordered::new();
417        // The topics with a stream that the loop reads, and the streams that wait for them.
418        let mut waiting: HashMap<TopicId, VecDeque<TopicReader>> = HashMap::new();
419        // The streams of the peer that ended, and how many it opened, once it said so.
420        let mut ended = 0u64;
421        let mut peer_opened = None::<u64>;
422
423        // Reads the `GOAWAY`, then ends when the peer finished the stream: it read all of ours.
424        // This future owns `ctl` while the loop uses `self`, so `ctl` is not a field of `self`.
425        let (goaway_tx, mut goaway_rx) = oneshot::channel();
426        let max_message_size = self.max_message_size;
427        let mut control = Box::pin(async move {
428            let ControlMessage::Goaway(n) =
429                ControlMessage::read(&mut ctl, &mut Vec::new(), max_message_size)
430                    .await
431                    .anyerr()?;
432            debug!(n, "GOAWAY received");
433            goaway_tx.send(n).ok();
434            ctl.read_to_end(0).await.anyerr().map(drop)
435        });
436        let mut control_finished = false;
437
438        let conn = self.conn.clone();
439        let closed = conn.closed();
440        tokio::pin!(closed);
441        let mut conn_closed = false;
442
443        // After a close, the loop waits for the header and the reads that are under way.
444        while !conn_closed || header.is_some() || !reads.is_empty() {
445            // A header that the loop still reads counts as an open stream. It times out
446            // after `HEADER_TIMEOUT`.
447            let streams_open = header.is_some() || !waiting.is_empty();
448            set_if_changed(&self.signals.peer_streams_open, streams_open);
449
450            tokio::select! {
451                _ = &mut closed, if !conn_closed => conn_closed = true,
452
453                // A new stream of the peer. We accept the next one after its header.
454                stream = self.conn.accept_uni(), if !conn_closed && header.is_none() => match stream {
455                    Ok(stream) => header = Some(Box::pin(TopicReader::accept(stream, self.max_message_size))),
456                    Err(_) => conn_closed = true,
457                },
458
459                // The header of a new stream: verify it, then read the stream, or let it wait.
460                reader = async { header.as_mut().expect("guarded").await }, if header.is_some() => {
461                    header = None;
462                    let mut reader = reader.map_err(|err| Failure::protocol(format!("stream header: {err:#}")))?;
463                    let topic = reader.header.topic;
464                    debug!(topic=%topic.fmt_short(), "stream opened");
465                    if self.verify(&mut reader) == Verdict::Reject {
466                        // A stopped stream ends here.
467                        ended += 1;
468                        self.events.send(ConnEventKind::Rejected(topic)).await?;
469                    } else if let Some(queue) = waiting.get_mut(&topic) {
470                        queue.push_back(reader);
471                    } else {
472                        waiting.insert(topic, VecDeque::new());
473                        reads.push(Box::pin(reader.read_next()));
474                    }
475                }
476
477                // A message of a stream, or its end.
478                Some((reader, res)) = reads.next(), if !reads.is_empty() => {
479                    let topic = reader.header.topic;
480                    match res {
481                        Ok(Some(msg)) => {
482                            self.events.send(ConnEventKind::Message(msg)).await?;
483                            reads.push(Box::pin(reader.read_next()));
484                            continue;
485                        }
486                        Ok(None) => debug!(topic=%topic.fmt_short(), "stream closed"),
487                        Err(err) => {
488                            // A close fails each read. Its code decides how the connection ended.
489                            if self.conn.close_reason().is_some() {
490                                conn_closed = true;
491                                continue;
492                            }
493                            return Err(Failure::protocol(format!("topic stream: {err:#}")));
494                        }
495                    }
496                    // The stream ended. The loop now reads the next stream of its topic, if any.
497                    ended += 1;
498                    match waiting.get_mut(&topic).and_then(VecDeque::pop_front) {
499                        Some(next) => reads.push(Box::pin(next.read_next())),
500                        None => {
501                            waiting.remove(&topic);
502                        }
503                    }
504                }
505
506                Ok(n) = &mut goaway_rx, if !goaway_rx.is_terminated() => peer_opened = Some(n),
507
508                res = &mut control, if !control_finished => match res {
509                    Ok(()) => control_finished = true,
510                    // A close ends the stream, and its close code decides how the connection ended.
511                    Err(_) if self.conn.close_reason().is_some() => control_finished = true,
512                    Err(err) => return Err(Failure::protocol(format!("control stream: {err:#}"))),
513                },
514            }
515
516            // The peer retires, and we read all the streams that it counted: the writer answers.
517            let drained = peer_opened.is_some_and(|opened| ended >= opened);
518            if drained {
519                self.signals.recv_drained.notify_one();
520            }
521            // We read all the peer wrote, and the peer read all we wrote: close.
522            if drained && control_finished {
523                return Ok(match self.conn.close_reason() {
524                    None => {
525                        self.conn.close(CloseCode::Retired.code(), b"retired");
526                        ConnEnd::Retired
527                    }
528                    reason => end_of(reason),
529                });
530            }
531        }
532
533        debug!("recv loop closed");
534        Ok(end_of(self.conn.close_reason()))
535    }
536
537    /// Asks the verifier about a new stream, and stops the stream if it rejects.
538    fn verify(&mut self, reader: &mut TopicReader) -> Verdict {
539        let topic = reader.header.topic;
540        let verdict = if self.rejected_topics.contains(&topic) {
541            Verdict::Reject
542        } else {
543            // The loop needs the token only here.
544            let token = reader.header.token.take();
545            self.verifier
546                .verify(topic, self.conn.remote_id(), token.as_deref())
547        };
548        if verdict == Verdict::Reject {
549            debug!(topic=%topic.fmt_short(), "stream rejected, stop it");
550            // Past the cap, the loop does not remember the topic, and the verifier sees its
551            // streams again.
552            if self.rejected_topics.len() < MAX_REJECTED_TOPICS {
553                self.rejected_topics.insert(topic);
554            }
555            reader.stream.stop(STOP_REJECTED).ok();
556        }
557        verdict
558    }
559}

561/// A topic stream that we read: its header, and the buffer for its frames.
562#[derive(Debug)]
563struct TopicReader {
564    stream: RecvStream,
565    header: StreamHeader,
566    buffer: Vec<u8>,
567    max_message_size: usize,
568}
569
570impl TopicReader {
571    /// Reads the header of a new stream.
572    async fn accept(mut stream: RecvStream, max_message_size: usize) -> Result<Self, ReadError> {
573        let mut buffer = Vec::new();
574        let header = StreamHeader::read(&mut stream, &mut buffer).await?;
575        Ok(Self {
576            buffer,
577            max_message_size,
578            stream,
579            header,
580        })
581    }
582
583    /// Reads the next message of the stream, or `None` if it ended gracefully.
584    ///
585    /// Returns `self` together with the result, also when the read failed.
586    async fn read_next(mut self) -> ReadResult {
587        let msg = read_message(&mut self.stream, &mut self.buffer, self.max_message_size).await;
588        let topic = self.header.topic;
589        let msg = msg.map(|msg| msg.map(|message| ProtoMessage { topic, message }));
590        (self, msg)
591    }
592}

594/// Writes the queue of the peer on a connection, and retires the connection when it is idle.
595struct SendLoop {
596    conn: Connection,
597    events: EventSink,
598    signals: Arc<Signals>,
599    /// The send half of the control stream, for the `GOAWAY`.
600    ctl: SendStream,
601    retire: CancellationToken,
602    take_queue: Arc<Mailbox<TakeQueue>>,
603    shutdown: CancellationToken,
604    retire_after: Duration,
605    max_message_size: usize,
606    streams: HashMap<TopicId, SendStream>,
607    buffer: Vec<u8>,
608    /// A future per stream, which resolves once the peer stopped it or acknowledged all of it.
609    stream_ends: FuturesUnordered<BoxFuture<(TopicId, StreamId, Option<VarInt>)>>,
610    /// The topics for which the writer opens a stream when it has the queue.
611    topics_to_open: Vec<TopicToken>,
612    /// How many topic streams the writer opened on this connection.
613    opened_count: u64,
614    /// The topics that the peer refused on this connection.
615    ///
616    /// We open no stream for such a topic, and drop its queued messages. A new
617    /// connection starts without refusals, so the writer tries the topic again there.
618    refused_topics: HashSet<TopicId>,
619    /// The time at which a retire fails the connection.
620    ///
621    /// The writer sets it to [`RETIRE_TIMEOUT`] from now when it stops writing. It then
622    /// writes nothing more.
623    retiring_until: Option<Instant>,
624}
625
626impl SendLoop {
627    /// Writes the queue that the actor gives it, and retires the connection when it is idle.
628    ///
629    /// Either writer retires when no stream was open in either direction and
630    /// the queue was empty for `retire_after`. It finishes its streams, and
631    /// sends `GOAWAY` with the number of streams it opened. It does the same
632    /// when the receive loop read all streams that the peer counted in its
633    /// `GOAWAY`. It then also finishes the control stream, and leaves its
634    /// queue for the next connection.
635    ///
636    /// After the `GOAWAY` the writer writes nothing. It tells the actor, which
637    /// dials for a later message.
638    ///
639    /// After the queue closed, the writer finishes its streams and waits for
640    /// the queue to open again. It never leaves a message unfinished, except
641    /// when it fails.
642    ///
643    /// The writer keeps the reader in `queue`, which outlives this future. The
644    /// reader goes back with the end of the connection, also after a retire.
645    /// So the next writer starts once the peer read all that this one wrote.
646    async fn run(mut self, queue: &mut Option<QueueReader>) -> Result<(), WriteError> {
647        let conn = self.conn.clone();
648        let closed = conn.closed();
649        tokio::pin!(closed);
650        let mut peer_streams_open = self.signals.peer_streams_open.subscribe();
651        let mut queue_closed: Option<watch::Receiver<bool>> = None;
652        let mut quiet_since: Option<Instant> = None;
653
654        loop {
655            let retiring = self.retiring_until.is_some();
656            let writing = queue.is_some() && !retiring;
657            // A close of the queue finishes our streams, if we have any.
658            let wants_close =
659                writing && (!self.streams.is_empty() || !self.topics_to_open.is_empty());
660            // Nothing is open or queued in either direction: the idle time runs.
661            let quiet = !retiring
662                && self.streams.is_empty()
663                && self.stream_ends.is_empty()
664                && self.topics_to_open.is_empty()
665                && queue.as_ref().is_none_or(|queue| queue.is_empty())
666                && !*peer_streams_open.borrow_and_update();
667            let retire_at =
668                quiet.then(|| *quiet_since.get_or_insert_with(Instant::now) + self.retire_after);
669            if !quiet {
670                quiet_since = None;
671            }
672
673            tokio::select! {
674                biased;
675
676                _ = &mut closed => break,
677
678                // The connection lost a crossed dial: the same retire as for an idle one.
679                _ = self.retire.cancelled(), if !retiring => self.retire().await?,
680
681                TakeQueue { reader, topics } = self.take_queue.take() => {
682                    // The new queue replaces the old one. The old streams end first.
683                    self.streams.clear();
684                    queue_closed = Some(reader.closed.clone());
685                    *queue = Some(reader);
686                    // A writer that retires keeps the reader, and writes nothing.
687                    if !retiring {
688                        self.topics_to_open = topics;
689                    }
690                }
691
692                // The peer retires and we read all it sent: answer, and finish the stream, so
693                // that the peer may close. This arm comes before the queue, so that a busy queue
694                // does not delay a retire.
695                _ = self.signals.recv_drained.notified() => {
696                    self.retire().await?;
697                    self.ctl.finish().ok();
698                }
699
700                // This arm comes before the queue. The queue of a busy peer is always ready,
701                // and would keep the loop from collecting ended streams.
702                Some((topic, id, stop)) = self.stream_ends.next(), if !self.stream_ends.is_empty() => {
703                    self.on_stream_end(topic, id, stop).await?;
704                }
705
706                // The next message of the queue, or the end of the queue.
707                more = async {
708                    match (&mut *queue, writing) {
709                        (Some(queue), true) => queue.wait_next().await,
710                        _ => std::future::pending().await,
711                    }
712                } => if more {
713                    let queue = queue.as_mut().expect("a message came from the queue");
714                    self.write_current(queue).await?;
715                } else {
716                    // The actor dropped the queue.
717                    self.streams.clear();
718                    *queue = None;
719                },
720
721                // These come after the queue, so they only run when the queue is empty.
722                _ = self.shutdown.cancelled() => {
723                    self.stop_writing();
724                    break;
725                }
726
727                // We finish our streams and keep waiting on this connection, so the next
728                // message does not need a dial. Dropping a stream finishes it.
729                _ = async {
730                    match &mut queue_closed {
731                        Some(queue_closed) if wants_close => {
732                            queue_closed.wait_for(|closed| *closed).await.ok();
733                        }
734                        _ => std::future::pending().await,
735                    }
736                } => {
737                    self.drop_streams();
738                }
739
740                // After the queue, so that queued messages get the streams of the peer
741                // first. A quiet topic needs a stream to keep the connection.
742                Some((topic, token)) = async {
743                    if writing { self.topics_to_open.pop() } else { None }
744                } => match self.open_stream(topic, token).await {
745                    // At the stream limit of the peer, the quiet topics get no stream.
746                    Err(WriteError::OpenStalled { .. }) => self.topics_to_open.clear(),
747                    Err(err) => tolerate_refusal(topic, err)?,
748                    Ok(()) => {}
749                },
750
751                // The connection was idle for `retire_after`.
752                () = time::sleep_until(retire_at.unwrap_or_else(Instant::now)), if retire_at.is_some() => self.retire().await?,
753
754                // The retire did not finish in time.
755                () = time::sleep_until(self.retiring_until.unwrap_or_else(Instant::now)), if retiring => return Err(n0_error::e!(WriteError::Stalled)),
756
757                // The peer opened or ended a stream, which restarts the idle time.
758                _ = peer_streams_open.changed() => {}
759            }
760        }
761
762        // Finish the streams, and wait until the peer acknowledged or stopped each one.
763        self.streams.clear();
764        while self.stream_ends.next().await.is_some() {}
765        // The writer ended without a close: the node shuts down. If the peer closed the
766        // connection, its close code stays.
767        if conn.close_reason().is_none() {
768            conn.close(CloseCode::Shutdown.code(), b"shutdown");
769        }
770        debug!("send loop closed");
771        Ok(())
772    }
773
774    /// Handles a stream that the peer stopped, or acknowledged in full.
775    async fn on_stream_end(
776        &mut self,
777        topic: TopicId,
778        id: StreamId,
779        stop: Option<VarInt>,
780    ) -> Result<(), WriteError> {
781        // Only the reject code is expected. Any other stop fails the connection.
782        let rejected = match stop {
783            Some(code) if code != STOP_REJECTED => {
784                return Err(n0_error::e!(WriteError::Stopped {
785                    code: code.into_inner()
786                }));
787            }
788            stop => stop.is_some(),
789        };
790        // The next message of the topic opens a new stream.
791        if self
792            .streams
793            .get(&topic)
794            .is_some_and(|stream| stream.id() == id)
795        {
796            self.streams.remove(&topic);
797        }
798        // The peer refuses the topic, not the connection. Tell the
799        // protocol once, and send nothing more for the topic.
800        if rejected && self.refused_topics.insert(topic) {
801            warn!(topic = %topic.fmt_short(), "the peer refused the topic");
802            self.streams.remove(&topic);
803            self.topics_to_open.retain(|(t, _)| *t != topic);
804            self.events.send(ConnEventKind::Refused(topic)).await.ok();
805        }
806        Ok(())
807    }
808
809    /// Stops writing and sends `GOAWAY`, only on the first call.
810    async fn retire(&mut self) -> Result<(), WriteError> {
811        if self.retiring_until.is_some() {
812            return Ok(());
813        }
814        self.stop_writing();
815        debug!(opened = self.opened_count, "send GOAWAY");
816        write_all(
817            &mut self.ctl,
818            &ControlMessage::Goaway(self.opened_count).to_frame(),
819        )
820        .await?;
821        // The actor never waits for a send loop, so this does not deadlock.
822        self.events.send(ConnEventKind::Retiring).await.ok();
823        Ok(())
824    }
825
826    /// Finishes the streams, and opens none of the pending ones.
827    fn drop_streams(&mut self) {
828        self.streams.clear();
829        self.topics_to_open.clear();
830    }
831
832    /// Finishes the streams, and writes nothing more.
833    fn stop_writing(&mut self) {
834        self.drop_streams();
835        self.retiring_until = Some(Instant::now() + RETIRE_TIMEOUT);
836    }

838    /// Writes the message that the queue has ready.
839    async fn write_current(&mut self, queue: &mut QueueReader) -> Result<(), WriteError> {
840        let item = queue.current().expect("wait_next returned a message");
841        let topic = item.message.topic;
842        if self.refused_topics.contains(&topic) {
843            trace!(topic = %topic.fmt_short(), "the peer refused the topic, drop the message");
844            // The protocol took the peer back, for example after a second `Join`.
845            let message = &item.message.message;
846            let readded = message.is_membership() && !message.is_disconnect();
847            queue.advance();
848            if readded {
849                self.events.send(ConnEventKind::Refused(topic)).await.ok();
850            }
851            return Ok(());
852        }
853        if let Err(err) = self.open_stream(topic, item.token.clone()).await {
854            tolerate_refusal(topic, err)?;
855            queue.advance();
856            return Ok(());
857        }
858        self.send(item).await?;
859        // A topic that we just left needs no stream.
860        if item.message.message.is_disconnect() {
861            self.topics_to_open.retain(|(t, _)| *t != topic);
862        }
863        queue.advance();
864        Ok(())
865    }
866
867    /// Writes a message on the stream of its topic, and finishes the stream after a `Close`.
868    ///
869    /// If the write fails because the peer refused the stream, the message is lost.
870    /// Returns the error if the write failed otherwise, which fails the connection.
871    async fn send(&mut self, item: &QueuedMessage) -> Result<(), WriteError> {
872        let ProtoMessage { topic, message } = &item.message;
873        let stream = self
874            .streams
875            .get_mut(topic)
876            .expect("the stream was opened before");
877        let len = item.encoded_len();
878        match write_sized_message(
879            stream,
880            message,
881            len,
882            &mut self.buffer,
883            self.max_message_size,
884        )
885        .await
886        {
887            Ok(()) => {
888                if message.is_disconnect() {
889                    // Dropping a stream finishes it.
890                    self.streams.remove(topic);
891                }
892                Ok(())
893            }
894            // Refused before any byte went out, so the stream is intact.
895            Err(WriteError::TooLarge { .. }) => {
896                warn!(max = self.max_message_size, "message too large, dropped");
897                Ok(())
898            }
899            Err(err) => {
900                tolerate_refusal(*topic, err)?;
901                self.streams.remove(topic);
902                Ok(())
903            }
904        }
905    }
906
907    /// Opens the stream of `topic` and writes its header, unless it is open.
908    ///
909    /// The header goes out in full once the stream opened. If the actor aborts the
910    /// peer, the connection closes anyway and cuts the header.
911    async fn open_stream(
912        &mut self,
913        topic: TopicId,
914        token: Option<Bytes>,
915    ) -> Result<(), WriteError> {
916        let Entry::Vacant(entry) = self.streams.entry(topic) else {
917            return Ok(());
918        };
919        let mut stream = time::timeout(OPEN_TIMEOUT, self.conn.open_uni())
920            .await
921            .map_err(|_| n0_error::e!(WriteError::OpenStalled))??;
922        StreamHeader::write(topic, token.as_deref(), &mut stream, &mut self.buffer).await?;
923        self.opened_count += 1;
924        debug!(topic=%topic.fmt_short(), "stream opened");
925        // The QUIC stack of the peer acknowledges data that its app did not read
926        // yet, so this needs no timeout.
927        let (id, stopped) = (stream.id(), stream.stopped());
928        self.stream_ends.push(Box::pin(async move {
929            let stop = stopped.await.ok().flatten();
930            (topic, id, stop)
931        }));
932        entry.insert(stream);
933        Ok(())
934    }
935}
936
937/// Returns the error if the write failed in a way that fails the connection.
938///
939/// A stream that the peer stopped with the reject code only loses its message.
940fn tolerate_refusal(topic: TopicId, err: WriteError) -> Result<(), WriteError> {
941    match &err {
942        WriteError::Stopped { code, .. } if *code == STOP_REJECTED.into_inner() => {
943            debug!(topic = %topic.fmt_short(), "the peer refused the stream, drop the message");
944            Ok(())
945        }
946        _ => Err(err),
947    }
948}
