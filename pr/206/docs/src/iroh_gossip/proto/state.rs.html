65/// A timer to be registered into the runtime
66///
67/// As the implementation of the protocol is an IO-less state machine, registering timers does not
68/// happen within the protocol implementation. Instead, these `Timer` structs are emitted as
69/// [`OutEvent`]s. The implementer must register the timer in its runtime to be emitted on the specified [`Instant`],
70/// and once triggered inject an [`InEvent::TimerExpired`] into the protocol state.
71#[derive(Clone, Debug)]
72pub struct Timer<PI> {
73    topic: TopicId,
74    /// The instance of the topic that armed the timer.
75    instance: u64,
76    timer: topic::Timer<PI>,
77}
78
79impl<PI> Timer<PI> {
80    /// Returns whether this timer runs a periodic membership task.
81    #[cfg(any(test, feature = "test-utils"))]
82    pub(crate) fn is_membership_cycle(&self) -> bool {
83        matches!(
84            self.timer,
85            topic::Timer::Swarm(
86                super::hyparview::Timer::DoShuffle | super::hyparview::Timer::DoMaintenance
87            )
88        )
89    }
90}

92/// Input event to the protocol state.
93#[derive(Clone, Debug)]
94#[non_exhaustive]
95pub enum InEvent<PI> {
96    /// Message received from the network.
97    RecvMessage(PI, Message<PI>),
98    /// Execute a command from the application.
99    Command(TopicId, Command<PI>),
100    /// Trigger a previously scheduled timer.
101    TimerExpired(Timer<PI>),
102    /// Peer disconnected on the network level.
103    PeerDisconnected(PI),
104    /// Update the opaque peer data about yourself.
105    UpdatePeerData(PeerData),
106    /// The network dropped a message to the peer, as the queue to it was full.
107    ///
108    /// Plumtree announces a dropped broadcast to the peer later, so that the
109    /// peer can request it.
110    SendDropped(PI, Message<PI>),
111}
112
113/// Output event from the protocol state.
114#[derive(Debug, Clone)]
115#[non_exhaustive]
116pub enum OutEvent<PI> {
117    /// Send a message on the network
118    SendMessage(PI, Message<PI>),
119    /// Emit an event to the application.
120    EmitEvent(TopicId, topic::Event<PI>),
121    /// Schedule a timer. The runtime is responsible for sending an [InEvent::TimerExpired]
122    /// after the duration.
123    ScheduleTimer(Duration, Timer<PI>),
124    /// Close the connection to a peer on the network level.
125    DisconnectPeer(PI),
126    /// Updated peer data
127    PeerData(PI, PeerData),
128}
129
130type ConnsMap<PI> = HashMap<PI, HashSet<TopicId>>;
131
132/// How many answers to messages for topics that we are not on [`State`] remembers at most.
133const MAX_TOLD_LEFT: usize = 4096;
134type Outbox<PI> = Vec<OutEvent<PI>>;
135
136enum InEventMapped<PI> {
137    All(topic::InEvent<PI>),
138    TopicEvent(TopicId, topic::InEvent<PI>),
139}
140
141impl<PI> From<InEvent<PI>> for InEventMapped<PI> {
142    fn from(event: InEvent<PI>) -> InEventMapped<PI> {
143        match event {
144            InEvent::RecvMessage(from, Message { topic, message }) => {
145                Self::TopicEvent(topic, topic::InEvent::RecvMessage(from, message))
146            }
147            InEvent::Command(topic, command) => {
148                Self::TopicEvent(topic, topic::InEvent::Command(command))
149            }
150            InEvent::TimerExpired(Timer { topic, timer, .. }) => {
151                Self::TopicEvent(topic, topic::InEvent::TimerExpired(timer))
152            }
153            InEvent::PeerDisconnected(peer) => Self::All(topic::InEvent::PeerDisconnected(peer)),
154            InEvent::UpdatePeerData(data) => Self::All(topic::InEvent::UpdatePeerData(data)),
155            InEvent::SendDropped(peer, Message { topic, message }) => {
156                Self::TopicEvent(topic, topic::InEvent::SendDropped(peer, message))
157            }
158        }
159    }
160}

162/// The state of the `iroh-gossip` protocol.
163///
164/// The implementation works as an IO-less state machine. The implementer injects events through
165/// [`Self::handle`], which returns an iterator of [`OutEvent`]s to be processed.
166///
167/// This struct contains a map of [`topic::State`] for each topic that was joined. It mostly acts as
168/// a forwarder of [`InEvent`]s to matching topic state. Each topic's state is completely
169/// independent; thus the actual protocol logic lives with [`topic::State`].
170#[derive(Debug)]
171pub struct State<PI, R> {
172    me: PI,
173    me_data: PeerData,
174    config: Config,
175    rng: R,
176    states: HashMap<TopicId, topic::State<PI, R>>,
177    outbox: Outbox<PI>,
178    peer_topics: ConnsMap<PI>,
179    /// The instance of each joined topic, so that the timers of a former one do nothing.
180    instances: HashMap<TopicId, u64>,
181    next_instance: u64,
182    /// The peers that we told that we are not on a topic, so that each hears it once.
183    ///
184    /// Messages that were in flight when we left a topic would each get an answer otherwise.
185    told_left: HashSet<(PI, TopicId)>,
186}
187
188impl<PI: PeerIdentity, R: Rng + SeedableRng> State<PI, R> {
189    /// Create a new protocol state instance.
190    ///
191    /// `me` is the [`PeerIdentity`] of the local node, `peer_data` is the initial [`PeerData`]
192    /// (which can be updated over time).
193    /// For the protocol to perform as recommended in the papers, the [`Config`] should be
194    /// identical for all nodes in the network.
195    ///
196    /// ## Panics
197    ///
198    /// Panics if [`Config::max_message_size`] is below [`MIN_MAX_MESSAGE_SIZE`].
199    pub fn new(me: PI, me_data: PeerData, config: Config, rng: R) -> Self {
200        assert!(
201            config.max_message_size >= MIN_MAX_MESSAGE_SIZE,
202            "max_message_size must be at least {MIN_MAX_MESSAGE_SIZE}"
203        );
204        Self {
205            me,
206            me_data,
207            config,
208            rng,
209            states: Default::default(),
210            outbox: Default::default(),
211            peer_topics: Default::default(),
212            instances: Default::default(),
213            next_instance: 0,
214            told_left: Default::default(),
215        }
216    }

279    /// Handle an [`InEvent`]
280    ///
281    /// This returns an iterator of [`OutEvent`]s that must be processed.
282    pub fn handle(
283        &mut self,
284        event: InEvent<PI>,
285        now: Instant,
286        metrics: Option<&Metrics>,
287    ) -> impl Iterator<Item = OutEvent<PI>> + '_ + use<'_, PI, R> {
288        trace!("in : {event:?}");
289        if let Some(metrics) = &metrics {
290            track_in_event(&event, metrics);
291        }
292
293        // A timer of a topic we quit, or of its former instance, is stale.
294        if let InEvent::TimerExpired(timer) = &event {
295            if self.instances.get(&timer.topic) != Some(&timer.instance) {
296                return self.outbox.drain(..);
297            }
298        }
299        let event: InEventMapped<PI> = event.into();
300
301        match event {
302            InEventMapped::TopicEvent(topic, event) => {
303                // when receiving a join command, initialize state if it doesn't exist
304                if matches!(&event, topic::InEvent::Command(Command::Join(_peers))) {
305                    if let hash_map::Entry::Vacant(e) = self.states.entry(topic) {
306                        self.told_left.retain(|(_, left)| *left != topic);
307                        self.next_instance += 1;
308                        self.instances.insert(topic, self.next_instance);
309                        e.insert(topic::State::with_rng(
310                            self.me,
311                            Some(self.me_data.clone()),
312                            self.config.clone(),
313                            R::from_rng(&mut self.rng),
314                        ));
315                    }
316                }
317
318                // when receiving a quit command, note this and drop the topic state after
319                // processing this last event
320                let quit = matches!(event, topic::InEvent::Command(Command::Quit));
321
322                // pass the event to the state handler
323                if let Some(state) = self.states.get_mut(&topic) {
324                    // A HyParView message makes the topic use the peer. A Plumtree message
325                    // does not: it can be a straggler from a peer that we dropped.
326                    if let topic::InEvent::RecvMessage(from, topic::Message::Swarm(_)) = &event {
327                        self.peer_topics.entry(*from).or_default().insert(topic);
328                    }
329                    let out = state.handle(event, now);
330                    for event in out {
331                        handle_out_event(topic, event, &mut self.peer_topics, &mut self.outbox);
332                    }
333                } else if let topic::InEvent::RecvMessage(from, message) = &event {
334                    // The sender thinks that we are on the topic, as after a restart of
335                    // ours. Tell it that we left.
336                    if self.told_left.len() >= MAX_TOLD_LEFT {
337                        self.told_left.clear();
338                    }
339                    if !message.is_disconnect() && self.told_left.insert((*from, topic)) {
340                        let left = hyparview::Disconnect {
341                            alive: false,
342                            _respond: false,
343                        };
344                        let message = topic::Message::Swarm(hyparview::Message::Disconnect(left));
345                        self.outbox
346                            .push(OutEvent::SendMessage(*from, Message { topic, message }));
347                    }
348                }
349
350                if quit {
351                    self.states.remove(&topic);
352                    self.instances.remove(&topic);
353                    // A peer the topic only sent to, such as an unanswered join, has no
354                    // other way out.
355                    let mut unused = Vec::new();
356                    for (peer, topics) in self.peer_topics.iter_mut() {
357                        if topics.remove(&topic) && topics.is_empty() {
358                            unused.push(*peer);
359                        }
360                    }
361                    for peer in unused {
362                        self.peer_topics.remove(&peer);
363                        self.outbox.push(OutEvent::DisconnectPeer(peer));
364                    }
365                }
366            }
367            // when a peer disconnected on the network level, forward event to all states
368            InEventMapped::All(event) => {
369                if let topic::InEvent::UpdatePeerData(data) = &event {
370                    self.me_data = data.clone();
371                }
372                for (topic, state) in self.states.iter_mut() {
373                    let out = state.handle(event.clone(), now);
374                    for event in out {
375                        handle_out_event(*topic, event, &mut self.peer_topics, &mut self.outbox);
376                    }
377                }
378                // If the peer disconnected, make sure to clear its `peer_topics` entry here.
379                // `handle_out_event` does the same, but only for peers that we were neighbors
380                // with. Peers that only relayed a shuffle or forward join to us also have
381                // entries in `peer_topics`, so we clear them here explicitly. This has to
382                // stay after the loop: `handle_out_event` needs the entry to tell whether a
383                // topic's `DisconnectPeer` was the peer's last.
384                if let topic::InEvent::PeerDisconnected(peer) = &event {
385                    self.peer_topics.remove(peer);
386                    self.told_left.retain(|(told, _)| told != peer);
387                }
388            }
389        }
390
391        for event in self.outbox.iter_mut() {
392            if let OutEvent::ScheduleTimer(_, timer) = event {
393                timer.instance = self.instances.get(&timer.topic).copied().unwrap_or(0);
394            }
395        }
396
397        // track metrics
398        if let Some(metrics) = &metrics {
399            track_out_events(&self.outbox, metrics);
400        }
401
402        self.outbox.drain(..)
403    }
404}

406fn handle_out_event<PI: PeerIdentity>(
407    topic: TopicId,
408    event: topic::OutEvent<PI>,
409    conns: &mut ConnsMap<PI>,
410    outbox: &mut Outbox<PI>,
411) {
412    trace!("out: {event:?}");
413    match event {
414        topic::OutEvent::SendMessage(to, message) => {
415            // HyParView may send to a peer that is no neighbor yet. The topic uses it from then on.
416            if matches!(message, topic::Message::Swarm(_)) {
417                conns.entry(to).or_default().insert(topic);
418            }
419            outbox.push(OutEvent::SendMessage(to, Message { topic, message }))
420        }
421        topic::OutEvent::EmitEvent(event) => outbox.push(OutEvent::EmitEvent(topic, event)),
422        topic::OutEvent::ScheduleTimer(delay, timer) => {
423            // `State::handle` sets the instance.
424            let timer = Timer {
425                topic,
426                instance: 0,
427                timer,
428            };
429            outbox.push(OutEvent::ScheduleTimer(delay, timer))
430        }
431        topic::OutEvent::DisconnectPeer(peer) => {
432            // The connection is shared by every topic that uses the peer.
433            if let Some(topics) = conns.get_mut(&peer) {
434                topics.remove(&topic);
435                if topics.is_empty() {
436                    // If the peer is no longer used by any topic, disconnect.
437                    conns.remove(&peer);
438                    outbox.push(OutEvent::DisconnectPeer(peer));
439                }
440            }
441        }
442        topic::OutEvent::PeerData(peer, data) => outbox.push(OutEvent::PeerData(peer, data)),
443    }
444}

446fn track_out_events<PI: Serialize>(events: &[OutEvent<PI>], metrics: &Metrics) {
447    for event in events {
448        match event {
449            OutEvent::SendMessage(_to, message) => match message.kind() {
450                MessageKind::Data => {
451                    metrics.msgs_data_sent.inc();
452                    metrics
453                        .msgs_data_sent_size
454                        .inc_by(message.size().unwrap_or(0) as u64);
455                }
456                MessageKind::Control => {
457                    metrics.msgs_ctrl_sent.inc();
458                    metrics
459                        .msgs_ctrl_sent_size
460                        .inc_by(message.size().unwrap_or(0) as u64);
461                }
462            },
463            OutEvent::EmitEvent(_topic, event) => match event {
464                super::Event::NeighborUp(_peer) => {
465                    metrics.neighbor_up.inc();
466                }
467                super::Event::NeighborDown(_peer) => {
468                    metrics.neighbor_down.inc();
469                }
470                _ => {}
471            },
472            _ => {}
473        }
474    }
475}
476
477fn track_in_event<PI: Serialize>(event: &InEvent<PI>, metrics: &Metrics) {
478    if let InEvent::RecvMessage(_from, message) = event {
479        match message.kind() {
480            MessageKind::Data => {
481                metrics.msgs_data_recv.inc();
482                metrics
483                    .msgs_data_recv_size
484                    .inc_by(message.size().unwrap_or(0) as u64);
485            }
486            MessageKind::Control => {
487                metrics.msgs_ctrl_recv.inc();
488                metrics
489                    .msgs_ctrl_recv_size
490                    .inc_by(message.size().unwrap_or(0) as u64);
491            }
492        }
493    }

648            StdRng::seed_from_u64(0),
649        );
650        let topic: TopicId = [1; 32].into();
651        let join = || InEvent::Command(topic, Command::Join(vec![]));
652        // A shuffle reply names peer 2, and the refill sends it a `Neighbor`.
653        let reply = || {
654            let reply = hyparview::Message::test_shuffle_reply_with(vec![2]);
655            InEvent::RecvMessage(
656                1,
657                Message {
658                    topic,
659                    message: topic::Message::Swarm(reply),
660                },
661            )
662        };
663        let now = Instant::now();
664        let request_timer = |out: Vec<OutEvent<u32>>| {
665            out.into_iter()
666                .find_map(|event| match event {
667                    OutEvent::ScheduleTimer(_, timer)
668                        if matches!(
669                            &timer.timer,
670                            topic::Timer::Swarm(hyparview::Timer::PendingNeighborRequest(2, _))
671                        ) =>
672                    {
673                        Some(timer)
674                    }
675                    _ => None,
676                })
677                .expect("a neighbor request to peer 2")
678        };
679        handle(&mut state, join());
680        let old = request_timer(state.handle(reply(), now, None).collect());
681        handle(&mut state, InEvent::Command(topic, Command::Quit));
682        handle(&mut state, join());
683        let current = request_timer(state.handle(reply(), now, None).collect());
684        let disconnects = |out: &[OutEvent<u32>]| {
685            out.iter()
686                .any(|event| matches!(event, OutEvent::DisconnectPeer(2)))
687        };
688
689        let out: Vec<_> = state
690            .handle(InEvent::TimerExpired(old), now, None)
691            .collect();
692        assert!(
693            !disconnects(&out),
694            "the old timer ended the new request: {out:?}"
695        );
696        let out: Vec<_> = state
697            .handle(InEvent::TimerExpired(current), now, None)
698            .collect();
699        assert!(disconnects(&out), "the request did not time out: {out:?}");
700    }
